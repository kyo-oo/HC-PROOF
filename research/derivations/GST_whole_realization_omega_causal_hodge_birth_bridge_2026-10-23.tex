\section{All-\(N\) whole-world causal triple-spectrum theorem}
\begin{theorem}[Full geometric Hodge/Chow/defect birth for all emergent dimensions]
Let \(X_0=X\) be any smooth projective complex variety and let
\(E_j\) be \emph{arbitrary actual algebraic} bundles of rank \(r_j\ge1\)
on \(X_{j-1}\). Form the genuine geometric world births
\[
 X_j=\mathbb P_{X_{j-1}}(\mathcal O\oplus E_j),
 \qquad d_j=d+\sum_{a=1}^jr_a,\qquad
 P_r(t)=1+t+\cdots+t^r.
\]
For every finite causal depth \(n\), with no upper bound on \(n\)
or on any rank, the \emph{simultaneous full-realization equations} are
\[
 \boxed{\mathcal H_{X_n}(u,v)=
       \mathcal H_X(u,v)\prod_{j=1}^n P_{r_j}(uv)},              \tag{3}
\]
\[
 \boxed{(\mathcal F_{X_n},\mathcal A_{X_n},\mathcal D_{X_n})(t)=
  (\mathcal F_X,\mathcal A_X,\mathcal D_X)(t)
           \prod_{j=1}^nP_{r_j}(t).}                          \tag{4}
\]
Thus all the born geometric worlds possess pure Hodge
decompositions, while each rational algebraic class and each
hypothetical rational nonalgebraic diagonal class has a
separately controlled causal channel.
\end{theorem}
\begin{proof}
For a projective bundle with relative dimension \(r_j\),
Leray--Hirsch gives
\(H^m(X_j,\mathbb Q)=
 \bigoplus_{i=0}^{r_j}\xi_j^i\pi_j^* H^{m-2i}(X_{j-1},\mathbb Q)\).
The tautological class \(\xi_j=c_1(\mathcal O(1))\) is algebraic
of pure Hodge type \((1,1)\), so the decomposition is of
pure rational Hodge structures. This gives (3) and the
\(\mathcal F\)-equation in (4). The corresponding genuine
Chow projective-bundle formula
\(CH^p(X_j)_{\mathbb Q}=
 \bigoplus_{i=0}^{r_j}\xi_j^i\pi_j^*CH^{p-i}(X_{j-1})_{\mathbb Q}\)
commutes with the canonical Betti cycle-class map.
It therefore gives \(\mathcal A_{X_j}=\mathcal A_{X_{j-1}}P_{r_j}\).
Taking their difference proves the third equation and induction
establishes arbitrary finite depth.
\end{proof}

\section{The \(\omega\)-causal survival theorem: unlimited finite births}
\begin{theorem}[Canonical first-defect conservation through all finite and \(\omega\) depths]
Suppose the rational defect \(D_{X,p_0}\ne0\) in the least such
codimension \(p_0\), if one exists. For every \(n\ge0\),
\[
 \boxed{D_{X,p_0}\ \xrightarrow{\sim}\
              D_{X_n,p_0}}                                   \tag{5}
\]
by the pullback class map. More generally the transition on
\(D_{X_n,p}\to D_{X_{n+1},p}\) is split injective by the
actual zero-section returns, so there is an injective
canonical map
\[
 \boxed{
 D_{X,p}\ \hookrightarrow\
 \underset{n<\omega}{\operatorname{colim}}\ D_{X_n,p}.}         \tag{6}
\]
The direct limit is formed in rational vector spaces for a
\emph{fixed codimension \(p\)}. It is not a claim that the
direct-limit space itself is a finite-dimensional smooth
projective variety or has a classical Hodge structure.
\end{theorem}
\begin{proof}
The projective Chow and Hodge decompositions imply
\(D_{X_{j},p}\cong
 \bigoplus_{i=0}^{r_j}D_{X_{j-1},p-i}\),
and the \(i=0\) inclusion is \(\pi_j^*\).
Its genuine algebraic zero section
\(s_j:X_{j-1}\to X_j\) satisfies
\(s_j^*\pi_j^*=\mathrm{id}\) simultaneously on Chow and
canonical Betti Hodge classes and therefore on defects.
So every stage transition is split injective.
For minimal \(p_0\), all \(i>0\) terms vanish, proving (5)
inductively. A vector from \(D_{X,p}\) maps to zero in the
direct limit only if it already vanishes after some
\emph{finite} stage; injectivity at every such stage
forces the original vector to be zero. This proves (6).
\end{proof}

\begin{corollary}[Exact high-dimension combinatorial amplification law]
If each emerging projective relative dimension is one, put
\(d_p(n)=\dim_{\mathbb Q}D_{X_n,p}\). Then
\[
 \boxed{d_p(n)=\sum_{k=0}^{\min(n,p)}
           {n\choose k}\,d_{p-k}(0).}                        \tag{7}
\]
For the least \(p_0\) with \(d_{p_0}(0)>0\), and each
fixed \(j\ge0\),
\[
 \boxed{d_{p_0+j}(n)=
       \frac{d_{p_0}(0)}{j!}n^j+O(n^{j-1})}
       \quad(n\to\infty),                                   \tag{8}
\]
with the \(j=0\) term exactly \(d_{p_0}(n)=d_{p_0}(0)\).
The leading coefficient is \emph{strictly positive} if
the hypothetical defect exists. A pure Hodge world can
therefore have arbitrarily large higher-codimension
canonical defect, even though none of its Hodge structure
is impure.
\end{corollary}
\begin{proof}
Equation (4) gives
\(\mathcal D_{X_n}(t)=\mathcal D_X(t)(1+t)^n\),
so the binomial theorem yields (7).
If \(p_0\) is least, then only the terms
\(d_{p_0},\ldots,d_{p_0+j}\) contribute to
\(d_{p_0+j}(n)\); the coefficient
\({n\choose j}d_{p_0}(0)\) has polynomial degree \(j\);
each other term has degree at most \(j-1\).
\end{proof}
