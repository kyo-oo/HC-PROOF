\section{The uniform kill (T1): boundary (d) discharged}

\begin{theorem}[Uniform kill]\label{th:uniform}
Let $1\to\GammaL\to G\to\Cfour\to 1$ be any extension and $A$ any
$G$-module. Then every class $c\in H^2(G;A)$ satisfies $16\,c=0$.
\end{theorem}

\begin{proof}
The Lyndon--Hochschild--Serre low-degree exact sequence reads
\[
H^2(\Cfour;A^\GammaL)\longrightarrow H^2(G;A)\longrightarrow
H^1\bigl(\Cfour;H^1(\GammaL;A)\bigr)\longrightarrow H^3(\Cfour;A^\GammaL).
\]
By exactness the kernel of the mixed-cell map is
$\im\bigl(H^2(\Cfour;A^\GammaL)\bigr)$, a quotient of
$H^2(\Cfour;A^\GammaL)$. The $|\Cfour|$-torsion law: for a finite group
$Q$ and any $Q$-module $M$, the composite
$\mathrm{cor}\circ\mathrm{res}=|Q|$ annihilates $H^n(Q;M)$ for $n\ge1$
(since $\mathrm{res}$ lands in $H^n(1;M)=0$), so $H^n(\Cfour;M)$ is
$4$-torsion. Hence $4\,\mathrm{im}(c)=0$ in the mixed layer, i.e.\
$\mathrm{mix}(4c)=0$, so $4c\in\ker(\mathrm{mix})$ which is itself
$4$-torsion: $16\,c=0$.
\end{proof}

\begin{corollary}[The input-axis denominator bound]
At every decked input --- whatever the honest $A^\GammaL$, whatever the
glue --- the rational chain of Round 28 closes with denominator at most
$16\cdot m\cdot(p-1)!$. The $2m$-versus-$4m$ distinction (boundary (d))
is an integral refinement, never a rational obstruction.
\end{corollary}

The certificate instantiates and gates the law on the module family:
every invariant factor of $H^n(\Cfour;M)$ ($n=1,2$) is $2$ or $4$ for
the Weil plane, the trivial module, the swap, the negation, and the
glue module $\Z[x]/((x+1)(x^2+1))$ (checks 0018--0031; the glue
realizes the $\Z/4$ at the mixed-layer level $H^1$, its deck layer
$H^2$ vanishing --- no eigenvalue-$1$ content --- which is exactly where
the $4m$ distinction lives). The group-level exponents are computed
directly by the integral total complex (Theorem~\ref{th:engine}): the
committed content $2$, the glue content $4$, the trivial content $4$
(checks 0039--0041).

\begin{remark}
The committed exponent $2$ is now computed \emph{directly}:
$H^2(\Gbul;A)\cong\Z^{12}\oplus(\Z/2)^6$ (Section~\ref{sec:sector}):
the torsion sector is the mixed layer's $(\Z/2)^6$ lifted, with no
$4$-torsion: the Round-28 SNF decision confirmed at the group level by
an independent engine.
\end{remark}

\section{The honest-invariant dichotomy (T2): boundary (b) discharged}

mixed-cell map $H^2(\Gbul;A)\to H^1(\Cfour;H^1(\GammaL;A))$ is exactly
$\im\bigl(H^2(\Cfour;A^\GammaL)\bigr)$. In particular the
concentration is injective whenever
$H^2(\Cfour;A^\GammaL)=0$ --- \emph{both} honest cases qualify:
\begin{enumerate}
\item[(i)] $A^\GammaL=A$ (the committed idealization): $H^2(\Cfour;A)=0$
by the committed $R1$ law (the norm degeneration
$N=(1+\gamma)(1+\gamma^2)=0$ with no invariants);
\item[(ii)] $A^\GammaL=0$ (the free case): constructed below.
\end{enumerate}
\end{theorem}

\begin{proof}

two cases make the image vanish. For (ii) the free module: the letters
$e_0,f_0$ act by the reflections $R_{e_0}=\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$
and $R_{f_0}=\left(\begin{smallmatrix}0&-1\\-1&0\end{smallmatrix}\right)$, the six fixed letters
trivially, and the deck by $S=\mathrm{diag}(1,-1)$; the eight letter

0043--0080). The fiber invariants
$A^\GammaL=\ker(R_{e_0}-1)\cap\ker(R_{f_0}-1)=\langle(1,1)\rangle\cap

\end{proof}

\section{The gerbe ambient (T3): $H^*(\Gbul;\F_2)$ exactly}

\begin{theorem}[The split total complex]\label{th:ambient}
With trivial $\F_2$-coefficients the Lyndon--Hochschild--Serre total
complex for $1\to\GammaL\to\Gbul\to\Cfour\to 1$ splits: the torus
product-CW differential vanishes, and over $\F_2$ the $\Cfour$-periodic
differentials reduce to $(1+\tau)$ and $0$ alternating, where
$\tau=\gamma\bmod 2$ is the transposition of $e_0,f_0$ fixing the six
letters. Consequently
\[
H^n(\Gbul;\F_2)\;=\;\bigoplus_{p+q=n}H^p\bigl(\Cfour;\wedge^q(\F_2^8)\bigr),
\qquad
H^0=1,\quad H^1=8,\quad H^2=30,\quad H^3=71.
\]
Since the $\mud$-coefficients of the torsion boundary are
$\F_2$-trivial, $H^3(\Gbul;\mud)\cong\F_2^{71}$: \emph{the ambient group
of the boundary class $\partial$ is computed exactly, and it is not
zero.}
\end{theorem}

\begin{proof}
The mod-$2$ letter action is the involution $\tau$; the letter space
decomposes as $P\oplus\F_2^6$ with $P$ the two-dimensional swap
module, so $\wedge^q(\F_2^8)\cong\F_2^{\binom6q+\binom6{q-2}}\oplus
P^{\binom6{q-1}}$, and both the invariants and the coinvariants of $P$
over $\F_2$ contribute one dimension each:
$H^p(\Cfour;\wedge^q)=\F_2^{d(q)}$ with
$d(q)=\binom6q+\binom6{q-1}+\binom6{q-2}$, giving the diagonal sums
$8,30,71$ (the closed-form gates 0015--0017). The split is verified
computationally: the total complex with
$d_{\mathrm{vert}}=0$ has the row cohomologies as its total cohomology,
and the hostile controls confirm the engine: the $H^1=8$ against the
direct abelianization count (the letter-coinvariant homs $1+6$ plus
the deck $1$; check 0011); the trivial deck against the binomial
diagonal $H^3=93$, $H^1=9$, $H^2=37$ (check 0012); the negation deck
($\F_2$-indistinguishable from the trivial deck --- the sign is
invisible mod $2$; check 0013); and the pure swap deck against the
committed deck (same ambient: the committed sign $e_0\mapsto -f_0$ is
invisible mod $2$; check 0014).
\end{proof}

\subsection{The $2\times2$ no-go}

\begin{theorem}[The content envelope cannot carry the deck]\label{th:nogo}
Realize the Weil plane by the traceless quaternion order
$W_R=\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$,
$W_I=\left(\begin{smallmatrix}0&-1\\1&0\end{smallmatrix}\right)$,
$W_K=W_RW_I=\left(\begin{smallmatrix}1&0\\0&-1\end{smallmatrix}\right)$
with $W_R^2=I$, $W_I^2=-I$, $W_RW_I=-W_IW_R$. The deck action
$\mathrm{CONT}$ on the content coordinates ($W_R\mapsto-W_I$,
$W_I\mapsto W_R$) is (i) not an algebra homomorphism of the order, and
(ii) not inner. Hence no multiplicative structure \emph{on the
$2\times2$ content plane} carries the committed deck: the nonabelian
reconstruction cannot live in the content envelope.
\end{theorem}

\begin{proof}
(i) The computed witness: $\mathrm{CONT}(W_I)=W_R$, so
$\mathrm{CONT}(W_I)^2=W_R^2=+I$, while the algebra relation demands the
image of $W_I^2=-I$ be $-I$ (check 0134). (ii) The unit group of the
order is $\{\pm I,\pm W_R,\pm W_I,\pm W_K\}$; every unit squares to
$\pm I$ (check 0135), so every inner automorphism has order $\le2$,
while $\mathrm{CONT}$ has order $4$.
\end{proof}

\begin{proof}
