\section{Theorem S1: the canonical middle splitting (strengthening Round 32, Theorem 1)}

Round 32 defined $M_{\mathrm{mid}}$ as the maximal level${}\le{}1$ rational
sub-Hodge-structure of $H^{2p-1}(X,\Q)$ and proved it two-level and
polarizable. This section strengthens the extraction to a \emph{canonical
splitting with an explicit description and a complement-independent
projector}.

\begin{theorem}[Canonical Middle Splitting]\label{thm:s1}
Let $X$ be a smooth projective complex variety, $p\ge1$, and write the
Lefschetz decomposition of the odd cohomology with respect to a hyperplane
class $\mathcal L$:
\[
H^{2p-1}(X,\Q)\;=\;\bigoplus_{j\ge0}\;\mathcal L^{j}\,P^{\,2p-1-2j}(X,\Q).
\]
Then:
\begin{enumerate}[label=(\roman*)]
\item \textbf{Explicit description.}
$M_{\mathrm{mid}}=\bigoplus_j \mathrm{mid}\bigl(\mathcal L^{j}
P^{\,2p-1-2j}\bigr)$, where $\mathrm{mid}(\cdot)$ denotes the sum of the
Hodge-type components with $|a-b|\le1$ (the types
$(p-j-1,\,p-j)$ and $(p-j,\,p-j-1)$ of each primitive piece). In particular
$M_{\mathrm{mid}}$ is a \emph{canonical direct summand}:
\[
H^{2p-1}(X,\Q)=M_{\mathrm{mid}}\oplus M_{\mathrm{high}},\qquad
M_{\mathrm{high}}:=\bigoplus_j \mathrm{high}\bigl(\mathcal L^{j}
P^{\,2p-1-2j}\bigr),
\]
with every Hodge type $(a,b)$ of $M_{\mathrm{high}}$ satisfying
$|a-b|\ge3$.
\item \textbf{The canonical projector.} The \emph{middle projector}
$\pi_{\mathrm{mid}}\in\End_{\Q,\mathrm{Hodge}}\bigl(H^{2p-1}(X)\bigr)$ ---
projection onto $M_{\mathrm{mid}}$ along $M_{\mathrm{high}}$ --- is
\emph{independent of every auxiliary choice}: any two $\Q$-rational Hodge
idempotents with image $M_{\mathrm{mid}}$ coincide. It is canonically
attached to the pair $(X,p)$.
\item \textbf{Nondegeneracy.} The polarization pairing of $H^{2p-1}(X)$
restricts nondegenerately to $M_{\mathrm{mid}}$; consequently
$M_{\mathrm{high}}=M_{\mathrm{mid}}^{\perp}$ for the standard polarization,
and the splitting of (i) is the orthogonal splitting.
\end{enumerate}
\PROVEN{}
\end{theorem}

\begin{proof}
(i) The hyperplane class has type $(1,1)$, so cupping with
$\mathcal L^{j}$ shifts a type $(a,b)$ to $(a+j,b+j)$ and preserves
$a-b$; the primitive pieces $P^{k}$ have $P^{a,b}\ne0$ only for $a+b=k$.
For $k=2p-1-2j$ odd, the types with $|a-b|\le1$ are exactly the two middle
types $(a,b)\in\{(p-j-1,\,p-j),(p-j,\,p-j-1)\}$. Each summand
$\mathrm{mid}(\mathcal L^{j}P^{k})$ is a $\Q$-sub-Hodge-structure (its set of
types is conjugation-stable, and the Lefschetz decomposition is defined over
$\Q$), of level $\le1$; hence it is contained in $M_{\mathrm{mid}}$ by
maximality. Conversely, if $v\in M_{\mathrm{mid}}$, decompose
$v=\sum_j\mathcal L^{j}v_j$ with $v_j$ primitive; each
$\mathcal L^{j}v_j$ is a $\Q$-rational component of a level${}\le{}1$
class, and since $\mathcal L^{j}$ preserves $a-b$, each $v_j$'s components
with $|a-b|\le1$ assemble into a sub-Hodge-structure summand of
$M_{\mathrm{mid}}$, while the $|a-b|\ge3$ components assemble into a
sub-Hodge-structure of level $\ge3$ contained in $M_{\mathrm{mid}}$ --- which
forces them to vanish (level${}\le1$ has no such types). So $v$ lies in the
right-hand side. The complement $M_{\mathrm{high}}$ is a sub-Hodge-structure
by the same type-stability, and its types satisfy $|a-b|\ge3$ because
$a+b=2p-1$ is odd, so $|a-b|$ is odd and $\ne1$.

(ii) Let $\pi,\pi'$ be $\Q$-rational Hodge idempotents with image
$M_{\mathrm{mid}}$. Then $\pi-\pi'$ vanishes on $M_{\mathrm{mid}}$ (both act
as the identity there), so it factors through
$\ker\pi\cap\ker\pi'$'s ambient complement; precisely,
$\pi-\pi'=(\pi-\pi')\circ(1-\pi)$, and $(1-\pi)$ has image
$M_{\mathrm{high}}$-typed (any Hodge idempotent with image $M_{\mathrm{mid}}$
has kernel with types disjoint from the middle types, by the same
sub-Hodge-structure argument as in (i)). A Hodge map sends type $(a,b)$ to
type $(a,b)$; since $M_{\mathrm{high}}$'s types and $M_{\mathrm{mid}}$'s
types are disjoint, $\Hom_{\mathrm{HS}}(M_{\mathrm{high}},M_{\mathrm{mid}})=0$,
so $\pi-\pi'=0$.

(iii) The polarization pairing $S$ on $H^{2p-1}$ is, on each summand
$\mathcal L^{j}P^{k}$, a nondegenerate pairing of that summand with itself
(Hodge--Riemann on the primitive piece, transported by
$\mathcal L^{j}$), and distinct Lefschetz summands are orthogonal for $S$.
Within $\mathcal L^{j}P^{k}$, $S$ pairs type $(a,b)$ against
$(k-a,k-b)$ only; for the middle types the pairing partner of
$(p-j-1,p-j)$ is $(p-j,p-j-1)$ --- both inside the middle sector, and the
Hodge--Riemann relations make it nondegenerate there. So
$S|_{M_{\mathrm{mid}}}$ is nondegenerate, $M_{\mathrm{mid}}\cap
M_{\mathrm{mid}}^{\perp}=0$, and dimension count gives the orthogonal
splitting; $M_{\mathrm{mid}}^{\perp}$'s types are then forced into
$|a-b|\ge3$ by the argument of (i), so $M_{\mathrm{mid}}^{\perp}=
M_{\mathrm{high}}$.
\end{proof}

\begin{remark}[What S1 changes]
Round 32's Theorem 1 asserted existence and uniqueness of
$M_{\mathrm{mid}}$ as a maximal sub-\emph{structure}. S1 makes the extraction
\emph{functorial-grade}: an explicit Lefschetz-piece formula, a canonical

\end{remark}

\section{Theorem S3: the unconditional parity wall at every input}

The campaign's parity wall (\S42 of the skill, ledger 095) was certified at
the committed worlds and at generic splits. Here it is proven
\emph{unconditionally, at every input}, in a form this session derived and
computed on its own (V8): the E-degree form.

\begin{theorem}[Unconditional Parity Wall]\label{thm:s3}
Let $E$ be the Gaussian elliptic curve, $X$ any smooth projective complex
variety. Define the \emph{E-degree} of a class in
$H^{*}(E\times X)$ as its Künneth component index in
$H^{d_E}(E)\otimes H^{*-d_E}(X)$ (so $d_E\in\{0,1,2\}$; a class is a sum of
its E-degree components).
\begin{enumerate}[label=(\roman*)]
\item Every integral $(1,1)$ class on $E\times X$ has E-degree-components
only in degrees $0$ and $2$: its E-degree-$1$ Künneth component is zero.
\item Consequently every class of the divisor algebra (the
$\Q$-span of finite products of integral $(1,1)$ classes) has
E-degree-$1$ Künneth component zero.
\item Every nonzero class of the balanced lattice
$B=\{x=e\otimes u+f\otimes v:\ u-iv\ \text{purely}(p-1,p)\}$
is pure of E-degree $1$. Hence
\[
B\;\cap\;\{\text{divisor algebra}\}\;=\;\{0\}
\qquad\text{at \emph{every} input }(X,p).
\]
\end{enumerate}
\PROVEN{}
\end{theorem}

\begin{proof}
(i) Let $c\in H^{2}(E\times X,\Z)\cap H^{1,1}$, with Künneth components
$c=c^{(0)}+c^{(1)}+c^{(2)}$ ($c^{(d)}\in H^{d}(E,\Z)\otimes H^{2-d}(X,\Z)$,
each integral). The component $c^{(1)}=e\otimes u+f\otimes v$ with
$u,v\in H^{1}(X,\Z)$: integrality means $u,v$ are fixed by conjugation
$\overline{(\cdot)}$ on $H^{1}(X,\C)$; the type condition means $e\otimes u$
is of type $(1,1)$, which forces $u$ purely of type $(0,1)$, and $f\otimes v$
of type $(1,1)$ forces $v$ purely $(1,0)$. But a conjugation-fixed vector
that is purely $(0,1)$ satisfies $u=\bar u\in H^{1,0}\cap H^{0,1}=0$
(conjugation maps $(0,1)$ to $(1,0)$), and likewise $v=0$. So $c^{(1)}=0$.
(ii) The Künneth components of a product are convolutions of the components
of the factors; a product of classes each with components only in E-degrees
$0$ and $2$ has E-degree-$1$ component
$\sum_{d+d'=1}c^{(d)}c'^{(d')}$ with $d,d'\in\{0,2\}$ --- an empty sum.
(iii) The balanced $x=e\otimes u+f\otimes v$ lies in
$H^{1}(E)\otimes H^{2p-1}(X)$ by construction: pure E-degree $1$; it is
nonzero iff $(u,v)\ne(0,0)$. Any class both in $B$ and in the divisor
algebra is pure E-degree $1$ with vanishing E-degree-$1$ component, hence
zero.
\end{proof}

\begin{remark}[The wall's correct scope, and the channel it leaves open]
The wall is a statement about the \emph{divisor algebra} --- products of
$(1,1)$ classes. It is unconditional and input-free. It does \emph{not}
close the mixed sector: the two-center conduit operates through
\emph{higher Chern classes of rank $\ge2$ bundles}, whose
$\mathrm{ch}^{p}$ components are pure $(p,p)$ but whose Künneth components
\emph{do} reach E-degree $1$ ($c_k$ for $k\ge2$ has an
$H^{1}(E)\otimes H^{2k-1}(X)$ part, and type-purity forces exactly the
balanced middle types). The wall certifies that no supply of the balanced
sector can come from divisors; the conduit channel and the correspondence
channel (Theorem S5) are the ones that can. \MINE{}: the finite shadow V8
computes the integrality-vs-type solve and the Künneth convolution.
\end{remark}

\section{Theorem S5: the divisor-twisted curve-channel supply (unconditional)}

This is the round's \emph{construction} theorem --- the first unconditional
supply of $K$-correspondences (and hence of algebraic balanced classes, via
Round 32's Theorem 2(b$\Rightarrow$a)) beyond $p=1$ and the committed
worlds.

\begin{theorem}[Divisor-Twisted Curve-Channel Supply]\label{thm:s5}
Let $X$ be smooth projective, $p\ge2$, $C$ a smooth projective curve with
Jacobian $J=J(C)$ of dimension $g$, and suppose given:
\begin{itemize}
\item an algebraic class $\delta\in\CH^{p-1}(X)_\Q$ (a divisor-word), and
\item an algebraic correspondence $V\in\CH^{1}(X\times C)_\Q$.
\end{itemize}
Then the composite
\[
K_{\delta,V}\;:=\;\Delta_{X,*}\bigl(\mathrm{cl}(\delta)\bigr)\;\circ\;V\;\circ\;
c_{1}(\mathcal P_{C})\;\circ\;\Delta_{J,*}\bigl(\mathcal L_{J}^{\,g-1}\bigr)
\;\in\;\CH^{\,p+g-1}(X\times J)_\Q
\]
is algebraic, where $\Delta_{X,*}(\mathrm{cl}\,\delta)$ is the
diagonal-correspondence acting as cup-product with
$\mathrm{cl}(\delta)$, $\mathcal P_{C}$ the Poincaré bundle on $C\times J$,
and $\mathcal L_J$ a hyperplane class on $J$. Its cohomology action on
$H^{1}(J)$ is, up to a nonzero rational scalar (the symplectic
normalization), the composite
\[
H^{1}(J)\;\xrightarrow[\ \cong\ ]{\ \text{canonical duality}\ }\;H^{1}(C)\;
\xrightarrow{\ V_*\ }\;H^{1}(X)\;\xrightarrow{\ \wedge\,\mathrm{cl}(\delta)\ }\;
H^{2p-1}(X),
\]
whose image lies automatically in $M_{\mathrm{mid}}(X,p)$ (type-chasing:
$V_*\beta$ has type $(1,0)$ or $(0,1)$, so the wedge with the
$(p-1,p-1)$ class $\delta$ has type $(p,p-1)$ or $(p-1,p)$).
Consequently $K_{\delta,V}$ is a $K$-correspondence for the sub-Hodge-structure
$\bigl[\,\mathrm{cl}(\delta)\wedge V_*H^{1}(C)\,\bigr]\subseteq M_{\mathrm{mid}}$,
and every balanced class $x_{u}$ with $u$ in this sub-structure is algebraic
on $E\times X$.
\PROVEN{}
\end{theorem}

\begin{proof}
\emph{The ingredients are algebraic:} $\delta$ is algebraic by hypothesis;
$\Delta_{X,*}\mathrm{cl}(\delta)\in\CH^{d+p-1}(X\times X)$ (pushforward of a
codimension-$(p-1)$ cycle along the codimension-$d$ diagonal) is the
correspondence acting as $\wedge\,\mathrm{cl}(\delta)$;
$V$ is algebraic by hypothesis;
$c_{1}(\mathcal P_{C})\in\CH^{1}(C\times J)$ is a divisor class;
$\Delta_{J,*}\mathcal L_{J}^{g-1}\in\CH^{2g-1}(J\times J)$ is the
correspondence acting as cup-$\mathcal L^{g-1}$ (hard Lefschetz,
$H^{1}\to H^{2g-1}$, both \STANDARD{}). Composites and sums of algebraic
correspondences are algebraic.

\emph{The codimension bookkeeping:} the composite
$[J\to C\to X\to X]$ has codimension
\[
\bigl(d+(p-1)\bigr)+1+\bigl(1+(g-1)\bigr)\;-\;d\;-\;1\;=\;p+g-1,
\]
subtracting the intermediate dimensions ($\dim X=d$, $\dim C=1$).
\CERT{} (V11 computes it on random instances, and the hostile
wrong-bookkeeping control fires.)

\emph{The action:} on $\beta\in H^{1}(J)$: cup-$\mathcal L^{g-1}$ gives
$\mathcal L^{g-1}\beta\in H^{2g-1}(J)$; the Poincaré-divisor correspondence
$c_{1}(\mathcal P_C)$ contracts it against the canonical duality: writing
$\{e_i\},\{f^{i}\}$ symplectic-dual letter bases of $H^{1}(C)$ and $H^{1}(J)$,

\COND{}\ (the $K$-hypothesis is the standing open condition; the

\end{theorem}

\begin{proof}

$\iota\circ\iota^{\ddagger}$ where $\iota\colon H^{1}(J)\cong
