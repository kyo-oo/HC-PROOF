\section{Dimensionless ontology and the full geometric realization}

identities and composition, without dimensions, metrics or Hodge groups.
\texttt{GSTGeneralSpaceRealization.lean} adds observed states;

\texttt{GSTGraphV2SelfExpandingCosmology.lean} and

codimension $p$, let
\[
 H_{X,p}=H^{2p}(X,\mathbb Q)\cap H^{p,p}(X),\qquad
 A_{X,p}=\operatorname{im}\bigl[
 CH^p(X)_{\mathbb Q}\xrightarrow{\mathrm{cl}^{can}_X}
 H^{2p}(X,\mathbb Q)\bigr],

\]

$\ker_{\mathbb Q}(G_X|_{H^{2p}})=H_{X,p}$.
This is a geometric spectral identity, not an assertion that
$A_{X,p}=H_{X,p}$.
The full rational Hodge multiplicity fiber of
\texttt{GSTClassicalHodgeFiberedCosmology.lean} must be retained.

The genuine graded GST cell signature $(k,a,b)$ defines

\section{GST whole-realization fold laws with a canonical cycle map}
Let an actual geometry-certified causal fold between smooth projective

   =\mathrm{cl}^{can}_Y\circ T_C.                   \tag{3}
\]

 K_T=G_YT-TG_X,\qquad K_{UT}=K_UT+UK_T.             \tag{4}

\begin{theorem}[Split causal fold algebraicity reflection for ALL emergent dimensions]
Suppose there are genuine return maps $S_H:H_{Y,p}\to H_{X,p}$
and $S_C:CH^p(Y)_{\mathbb Q}\to CH^p(X)_{\mathbb Q}$ with
\[
 S_H\mathrm{cl}^{can}_Y=\mathrm{cl}^{can}_X S_C,
 \qquad S_HT_H=\operatorname{id}_{H_{X,p}}.        \tag{5}
\]
Then $\bar S\bar T=\operatorname{id}_{D_{X,p}}$, hence
\[
 \boxed{\alpha\in A_{X,p}\ \Longleftrightarrow\
                  T_H\alpha\in A_{Y,p}},\qquad
 \boxed{D_{Y,p}\cong D_{X,p}\oplus\ker\bar S}.       \tag{6}
\]
Every finite composition of such causal folds, regardless of
how many geometric dimensions emerge, has the same property.
Importantly, the causal fold need not itself be reversible;
only its actual Hodge and canonical-Chow returns are required.
\end{theorem}
\begin{proof}
Equation (3) induces $\bar T$; the first equation of (5)
induces $\bar S$, and the second gives $\bar S\bar T=1$.
If $T_H\alpha=\mathrm{cl}^{can}_Y(Z_Y)$, then
$\mathrm{cl}^{can}_X(S_C Z_Y)=S_H T_H\alpha=\alpha$.
The converse transports any genuine source $Z_X$ with (3).
A split injection of rational vector spaces yields (6),
and composition of the two commuting squares and returns
proves the arbitrary finite-depth assertion by induction.
\end{proof}

\section{Exact all-$N$ topology/dimension-fold defect equation}
For arbitrary smooth projective $X_0=X$, recursively take
genuine projective folds
\[
 X_j=\mathbb P_{X_{j-1}}(\mathcal O\oplus E_j),
 \quad \operatorname{rank}(E_j)=r_j\ge1,
 \quad \dim X_n=\dim X+\sum_{j=1}^n r_j .
\]
No upper bound is imposed on $n$ or on $r_j$.
Projective-bundle Chow and rational Hodge formulas are
both compatible with the canonical cycle-class map.

\begin{theorem}[Complete all-depth emergent-dimensional defect spectrum]
For every $p\ge0$ and every finite fold-depth $n$,
\[
 \boxed{
 D_{X_n,p}\cong
 \bigoplus_{i_1=0}^{r_1}\cdots
 \bigoplus_{i_n=0}^{r_n}D_{X,p-i_1-\cdots-i_n}.
 }                                                     \tag{7}
\]
Set $d_p(X)=\dim_{\mathbb Q}D_{X,p}$ and
$\mathcal D_X(t)=\sum_{p\ge0}d_p(X)t^p$. Then
\[
 \boxed{\mathcal D_{X_n}(t)=
   \mathcal D_X(t)\prod_{j=1}^n(1+t+\cdots+t^{r_j}).}  \tag{8}
\]
If all $r_j=1$,
\[
 \boxed{d_p(X_n)=\sum_{k=0}^{\min(n,p)}
                   {n\choose k}d_{p-k}(X).}          \tag{9}
\]
If $p$ is the first nonzero-defect codimension on original
$X$, then $d_p(X_n)=d_p(X)$ for \emph{every} finite depth.
\end{theorem}
\begin{proof}
For each stage, the projective-bundle decomposition of
rational Hodge and rational Chow groups with powers of
the algebraic tautological class gives
$D_{X_j,p}\cong\oplus_{i=0}^{r_j}D_{X_{j-1},p-i}$.
Induction gives (7); multiplying finite generating
polynomials gives (8). The binomial identity
$(1+t)^n=\sum_k\binom nk t^k$ gives (9).
All summands for a first nonzero $p$ except
$i_1=\cdots=i_n=0$ vanish by minimality.
\end{proof}
