\section{Unconditional alpha-selected topological source}
\begin{theorem}[Original-\(X\) topological Chern source, every \(p\ge1\)]
For any \(\alpha\in H^{2p}(X,\Q)\), even without the Hodge condition, there are an integer \(m>0\), an integer \(r\ge\max(p,d)\), and an ACTUAL smooth complex bundle \(E_\alpha\) of rank \(r\) on \(X\) such that
\[
\ch(E_\alpha)=r+m\alpha\quad\text{in }H^{\mathrm{even}}(X,\Q),\qquad
c_i(E_\alpha)=0\ (1\le i<p),
\]
\[
\boxed{c_p(E_\alpha)=(-1)^{p-1}(p-1)!\,m\alpha.}\tag{1}
\]
Here the first displayed vanishing of lower \(c_i\)'s is a statement in \emph{rational cohomology}, not necessarily integrally.
\end{theorem}
\begin{proof}
The rational Chern character \(K^0_{\mathrm{top}}(X)\otimes\Q\xrightarrow{\sim}H^{\mathrm{even}}(X,\Q)\) is an isomorphism for the finite CW space \(X\). Pull back the \emph{single prescribed} degree-\(2p\) component \(m\alpha\), clearing denominators to a genuine integral virtual class \(v\). A smooth vector bundle has a smooth orthogonal complement in a trivial bundle, so after stabilization write \(v=[E_\alpha]-[\underline{\mathbb C}^N]\). Add trivial summands to ensure \(r\ge\max(p,d)\). All positive-degree Chern characters vanish except \(\ch_p(E_\alpha)=m\alpha\). Newton's identities give \(c_1=\cdots=c_{p-1}=0\) and \(c_p=(-1)^{p-1}(p-1)!\ch_p\) rationally.
\end{proof}

\section{Actual original-\(X\) determinantal carriers}
Fix an ample algebraic line bundle \(L\) with \(h=c_1(L)\). For each integer \(t>0\), put \(E_t=E_\alpha\otimes L^t\). Choose a generic \emph{smooth complex-linear} bundle morphism \(\phi_t:\underline{\mathbb C}^{\,r-p+1}\longrightarrow E_t\) transverse to the universal rank strata. Its rank-drop locus
\[
D_t=\{x\in X:\rk_{\mathbb C}\phi_t(x)\le r-p\}
\]
is a genuine compact, complex-oriented, real codimension-\(2p\) determinantal pseudocycle on the \emph{original} \(X\), smooth outside the deeper rank-drop stratum. The latter has expected real codimension \(4(p+1)\) in \(X\), sufficiently high to define a canonical top-dimensional integral homology class \([D_t]\). This construction is smooth/topological; \(\phi_t\) is not presumed holomorphic.

\begin{theorem}[Topological Porteous and universal twist polynomial]
For every \(t>0\),
\[
\boxed{\PD[D_t]=c_p(E_\alpha\otimes L^t)
=\sum_{i=0}^p {r-i\choose p-i}t^{p-i}h^{p-i}c_i(E_\alpha).}\tag{2}
\]
The polynomial in \(t\) has degree at most \(p\), and its constant term is \(c_p(E_\alpha)\).
\end{theorem}
\begin{proof}
Transversality to the complex determinantal stratification and the topological Thom--Porteous formula give \(c_{r-(r-p+1)+1}(E_t-\underline{\mathbb C}^{r-p+1})=c_p(E_t)\). By the splitting principle, if \(x_1,\ldots,x_r\) are formal Chern roots of \(E_\alpha\), then \(c_p(E_t)=e_p(x_1+th,\ldots,x_r+th)\). The coefficient of \(t^{p-i}h^{p-i}c_i(E_\alpha)\) is \(\binom{r-i}{p-i}\), proving (2).
\end{proof}

\]
Consequently the explicit rational combination
\[
\boxed{\mathcal S_{\alpha,p}^{\mathrm{smooth}}
=\frac{(-1)^{p-1}}{(p-1)!\,m}
\sum_{j=1}^{p+1}(-1)^{j-1}\binom{p+1}{j}[D_j]}

\boxed{\PD\mathcal S_{\alpha,p}^{\mathrm{smooth}}=\alpha.}\tag{4}

\boxed{\PD(3[D_1]-3[D_2]+[D_3])=c_2(E_\alpha)=-m\alpha.}\tag{5}

\section{A quantitative attempt at forcing holomorphic survival}
Write \(q=d-p\), let \(\omega\) be a K\"ahler form with \([\omega]=h\), and \(\Omega_q=\omega^q/q!\) the Wirtinger calibration. For \(M=1,2,\ldots\), repeat (3) on \emph{high twists} \(E_\alpha\otimes L^{jM}\), and write \(D_{jM}\) for independent generic smooth degeneracy pseudocycles. The moment identity still holds:
\[
\sum_{j=1}^{p+1}\lambda_j[D_{jM}]\ \text{has cohomology class}\ c_p(E_\alpha)
\quad\text{for every }M\ge1.
\]
\begin{proposition}[Proved mass-growth obstruction for uncorrelated world births]
Suppose the top-dimensional rectifiable mass measures of the \(D_{jM}\) are mutually singular (this is the case when the independently chosen generic degeneracy loci have no common top-dimensional components). Then, with \(T_M=\sum_j\lambda_j[D_{jM}]\),
\[
\boxed{
\Mass(T_M)-\int_{T_M}\Omega_q
\ \ge\
\binom rp\!\left(\sum_{j=1}^{p+1}|\lambda_j|j^p\right)
\left(\int_X\frac{h^d}{q!}\right)M^p-O(M^{p-1}).
}\tag{6}
\]
The leading coefficient is strictly positive. For \(p=2\) the moment factor is \(24\), for \(p=3\) it is \(224\). Therefore independent-world algebraic-charge cancellation DOES NOT give an almost calibrated geometric carrier; it requires actual \emph{correlated geometric support cancellation}.
\end{proposition}
\begin{proof}
Mutual singularity gives \(\Mass(T_M)=\sum_j|\lambda_j|\Mass(D_{jM})\). Wirtinger's comass inequality gives \(\Mass(D_{jM})\ge\int_{D_{jM}}\Omega_q\), even if a degeneracy locus is not holomorphic. The latter period equals \(\int_Xc_p(E_\alpha\otimes L^{jM})\,h^q/q!\), whose leading term is \(\binom rp(jM)^p\int_Xh^d/q!\) by (2). But the \emph{signed} period of \(T_M\) is the fixed \(\int_Xc_p(E_\alpha)\,h^q/q!\), again by (3). Subtracting gives (6). The coefficient is positive because each \(|\lambda_j|j^p>0\).
\end{proof}
