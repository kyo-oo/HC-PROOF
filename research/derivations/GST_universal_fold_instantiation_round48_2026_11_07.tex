\section{The input space at full scope: the phase quadraticity and the
four strata}

\begin{definition}[The two-letter world and its phase]\label{def:world}
A two-letter world carries the integral parameters $(s,q)$, $q>0$:
the letter action with trace $s$ and determinant $q$. Its
eigenvalues $\lambda,\mu$ are the roots of $x^2-sx+q$, and the
\emph{phase} is $z=\lambda/\mu$. Writing
$D_2=s^2-4q$ and $\sigma$ for the canonical conjugation of
$K=\Q(\sqrt{D_2})$:
\[
z=\frac{s+\sqrt{D_2}}{s-\sqrt{D_2}}
=\frac{s^2+D_2+2s\sqrt{D_2}}{s^2-D_2},
\qquad
z+\sigma(z)=\frac{s^2-2q}{q}=:t\in\Q .
\]
\end{definition}

\begin{lemma}[The phase quadraticity]\label{lem:quad}
Every two-letter world's phase satisfies
$z^2-tz+1=0$ with $t=z+\sigma(z)\in\Q$: the phase is quadratic over
$\Q$ (or rational). Moreover $z\,\sigma(z)=1$ identically --- the
polynomial identity $(s^2-D_2)^2/(s^2-D_2)^2=1$ --- at every input,
with no sign condition on $D_2$.
\end{lemma}

\begin{proof}
The characteristic law: $\lambda+\mu=s$, $\lambda\mu=q$, so
$z+z^{-1}=(\lambda^2+\mu^2)/(\lambda\mu)=((\lambda+\mu)^2-2\lambda\mu)/
(\lambda\mu)=(s^2-2q)/q=t$. Then $z^2-tz+1=0$ by multiplication by
$z$. The norm: $z\sigma(z)=\lambda\mu/(\mu\lambda)=1$. Both are
polynomial identities in $(s,q)$; the certificate computes them
exactly at every instance (layer X5).
\end{proof}
