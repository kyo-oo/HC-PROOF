\begin{abstract}
A dimension-free GST world can admit geometric realization maps without

mixed Hodge theory of singular realized worlds, \emph{not} a counterexample
to the rational Hodge conjecture for smooth projective varieties.
\end{abstract}

\section{General causal-graph source birth, not a label change}
Let $\Gamma$ be a finite connected undirected multigraph with
$r\geq1$ vertices, $e$ edges (loops and parallel edges allowed), and
\[
                  \beta=e-r+1\geq0.
\]
For each vertex take an independent copy of $\mathbb P^1_{\mathbb C}$.
For each edge choose two distinct marked smooth points, one on each
incident component (on the same component if a loop), and identify the
two points transversely to form an ordinary node.  The resulting
connected reduced projective nodal curve $C_\Gamma$ has normalization
\(\widetilde C_\Gamma=\coprod_{i=1}^r\mathbb P^1\).
The graph is the \emph{dual incidence graph of actual geometric
identifications}, not an arbitrary linear operator on Hodge labels.

\begin{theorem}[Causal graph cycles are emergent weight-zero holes]
For such $C_\Gamma$,
\[
 H^0(C_\Gamma,\mathbb Q)=\mathbb Q(0),\qquad
 H^1(C_\Gamma,\mathbb Q)\cong H^1(\Gamma,\mathbb Q)
       \cong\mathbb Q(0)^\beta,\qquad
 H^2(C_\Gamma,\mathbb Q)\cong\mathbb Q(-1)^r.
\]
The displayed $H^1$ is a pure weight-zero mixed Hodge summand in
cohomological degree one. It is born exactly when the causal graph has
a nontrivial cycle, not when it is a tree.
\end{theorem}
\begin{proof}
For the normalization $\nu:\widetilde C_\Gamma\to C_\Gamma$,
the difference of the two branch values at each node gives the exact
constant-sheaf sequence
\[
0\to\mathbb Q_{C_\Gamma}\to
\nu_*\mathbb Q_{\widetilde C_\Gamma}\to
\bigoplus_{\mathrm{node}\in E(\Gamma)}\mathbb Q_{\mathrm{node}}\to0.
\]
Since the normalized components are rational,
$H^1(\widetilde C_\Gamma,\mathbb Q)=0$.  The long exact sequence
contains
\[
0\to\mathbb Q\xrightarrow{\Delta}\mathbb Q^r
\xrightarrow{B_\Gamma}\mathbb Q^e
\to H^1(C_\Gamma,\mathbb Q)\to0,
\]
where $B_\Gamma$ is the oriented edge--vertex incidence matrix.
Connectedness makes $\operatorname{rank} B_\Gamma=r-1$ (a
spanning-tree argument, valid with loops), so the cokernel has rank
$e-r+1=\beta$.  This is also the graph cohomology $H^1(\Gamma,\mathbb Q)$.
The normalization sequence is compatible with mixed Hodge structures:
its graph quotient comes from weight-zero $H^0$ of points and
components, hence has weight zero and type $(0,0)$.
In degree two normalization gives $\mathbb Q(-1)^r$.
\end{proof}

\section{All-dimensional causal fold laws}
For every integer $N\geq0$ put
\[
       Y_{\Gamma,N}=C_\Gamma\times\mathbb P^N.
\]
This is an \emph{actual} projective complex realized world with complex
dimension $N+1$; it is singular when $\Gamma$ has any nodes.
Write $\mathcal H(t;u,v,s)$ for the degree--Hodge--weight polynomial,
where $t$ counts ordinary cohomological degree, $u^pv^q$ a Hodge
bidegree on $\operatorname{Gr}^W_{p+q}$, and $s$ the weight.
With $a=t^2s^2uv$, mixed-Hodge Kunneth yields the closed all-$N$ law
\[
\boxed{\quad
  \mathcal H(Y_{\Gamma,N};t,u,v,s)
  =(1+\beta t+r\,a)\sum_{j=0}^N a^j.
 \quad} \tag{1}
\]
In particular, at each $0\leq j\leq N$,
\[
\boxed{\ \operatorname{Gr}^{W}_{2j}
 H^{2j+1}(Y_{\Gamma,N},\mathbb Q)
 \cong\mathbb Q(-j)^\beta,\quad
 (p,q)=(j,j),\quad(\chi,\delta)=(0,-1).\ }\tag{2}
\]
Taking $t=-1,s=1$ gives independently the compact projective
Hodge--Deligne polynomial
$E(Y_{\Gamma,N};u,v)=(1-\beta+r\,uv)
\sum_{j=0}^N(uv)^j$.

\begin{theorem}[Two independent causal folds produce even-degree
charge-zero, non-pure classes]
Let $\Gamma_1,\Gamma_2$ be two connected finite multigraphs with
cycle ranks $\beta_1,\beta_2>0$, and with all normalization components
$\mathbb P^1$. Define
$Z_N=C_{\Gamma_1}\times C_{\Gamma_2}\times\mathbb P^N$.
For every $0\leq j\leq N$,
\[
\boxed{\quad
\operatorname{Gr}^{W}_{2j} H^{2j+2}(Z_N,\mathbb Q)
\cong
 H^1(\Gamma_1,\mathbb Q)\otimes
 H^1(\Gamma_2,\mathbb Q)(-j)
\cong\mathbb Q(-j)^{\beta_1\beta_2}.
\quad}\tag{3}
\]
Every nonzero class in this piece has
\[
    k=2j+2,\quad (p,q)=(j,j),\quad
    w=2j,\quad\chi=0,\quad\delta=-2.
\]
Thus the causal fold has created a genuinely rational, diagonal,
even-degree cohomological sector which the \emph{charge-only}
test accepts but the purity test excludes from the usual
smooth-projective Hodge conjecture.
\end{theorem}
\begin{proof}
The rational mixed-Hodge Kunneth decomposition contains
$H^1(C_{\Gamma_1})\otimes H^1(C_{\Gamma_2})
\otimes H^{2j}(\mathbb P^N)$ in degree $2j+2$.
Its weight is $0+0+2j=2j$, its type is $(j,j)$, and its rank is
$\beta_1\beta_2$. All other degree-$2j+2$ summands of
$C_{\Gamma_1}\times C_{\Gamma_2}\times\mathbb P^N$
have weight $2j+1$ or $2j+2$: each normalized rational-curve
factor contributes either $H^0$ of degree/weight $(0,0)$ or
$H^2$ of degree/weight $(2,2)$, while a graph $H^1$ contributes
$(1,0)$. Weight deficit $-2$ therefore occurs precisely in the
two-graph $H^1\otimes H^1$ term. This establishes the equality
of the weight graded piece, not merely a lower bound.
\end{proof}

\begin{corollary}[The phenomenon survives every arbitrary original $X$]

\begin{proof}

\end{proof}

\section{Reverse fold: smoothing restores pure Hodge geometry and births a native diagonal}
There is also a genuinely \emph{Hodge-preserving} outcome of
changing the same realized topology.  Let $\Gamma$ be connected and
all its normalization components rational, with cycle rank
$g=\beta\geq1$. Choose a flat projective one-parameter smoothing
$\mathscr C\to\Delta$ of the nodal curve $C_\Gamma$ whose general
fiber $C_t$ is a smooth projective connected curve.
The arithmetic genus formula for nodal curves gives
$p_a(C_\Gamma)=\sum_v g(\widetilde C_v)+\beta=g$; constancy
of arithmetic genus in the family gives $g(C_t)=g$.
(The theorem below quantifies over such an actual smoothing, rather
than declaring an abstract GST graph edge to be a smooth family.)

\begin{theorem}[Actual smoothing and the pure causal Hodge birth]
For every $N\geq0$, the smooth projective
$S_{t,N}=C_t\times\mathbb P^N$ has
\[
\boxed{\quad
\mathcal H(S_{t,N};t_0,u,v,s)=
\bigl(1+g\,t_0s(u+v)+t_0^2s^2uv\bigr)
\sum_{j=0}^{N}(t_0^2s^2uv)^j.
\quad}\tag{4}
\]
The nodal central fiber instead has
\[
\mathcal H(C_\Gamma\times\mathbb P^N)
=\bigl(1+g\,t_0+t_0^2s^2uv\bigr)
\sum_{j=0}^{N}(t_0^2s^2uv)^j
\]
when $\Gamma$ has one irreducible normalization component;
for general $r$ rational components its top-degree coefficient
is $r\,t_0^2s^2uv$ as in (1).
Consequently the smoothing creates a pure weight-$(2j+1)$
$H^{2j+1}$ sector of dimension $2g$ and types
$(j+1,j)$ and $(j,j+1)$, each multiplicity $g$.
It does \emph{not} turn the former weight-$2j$ classes into
degree-$(2j+1)$ Hodge classes by relabeling:
the geometric topology and cohomology change.
\end{theorem}
\begin{proof}
The smooth genus-$g$ curve has $H^0=\mathbb Q(0)$,
$H^1$ pure of weight one and Hodge numbers
$h^{1,0}=h^{0,1}=g$, and $H^2=\mathbb Q(-1)$.
Apply Kunneth and the projective-space Tate decomposition;
these prove (4) for all $N$ and all $j$.
The central-fiber formula is (1), with the correct number of
components $r$. For a nodal \emph{irreducible} rational curve
$r=1$ and $g$ self-nodes, the displayed contrast is literal.
\end{proof}

\begin{theorem}[Native algebraic middle diagonal born in the smooth world]
For each nonzero genus $g$ general fiber $C_t$, choose
a closed complex point $o\in C_t(\mathbb C)$ and set
\[
 Z_t=[\Delta_{C_t}]-[C_t\times\{o\}]
                   -[\{o\}\times C_t]
 \in CH^1(C_t\times C_t)_{\mathbb Q}.
\]
Its genuine canonical Betti class is nonzero, rational and
of pure type $(1,1)$, and lies in the
$H^1(C_t)\otimes H^1(C_t)$ Kunneth summand.
For every $0\leq j\leq N$ the actual algebraic cycle
$Z_t\times h^j$ on
$C_t\times C_t\times\mathbb P^N$
has nonzero rational pure class of type $(j+1,j+1)$,
weight and degree $2j+2$, hence $(\chi,\delta)=(0,0)$.
\end{theorem}
\begin{proof}
The class of the diagonal acts as the identity on $H^1(C_t)$
via the standard pull--cup--push correspondence.  The two
fiber correction terms act trivially on $H^1$, so the
$H^1\otimes H^1$ Kunneth component of $Z_t$ is the
nonzero identity correspondence.  The remaining $H^0\otimes H^2$
and $H^2\otimes H^0$ components are exactly removed by the
two fiber cycles. Every term is a genuine algebraic divisor
class on the smooth projective surface $C_t\times C_t$,
so its canonical Betti class is rational and $(1,1)$.
External cup product with the hyperplane cycle $h^j$
proves the higher-dimensional statement.
\end{proof}

\section{All-order GST causal emergence: dimensions and Hodge multiplicity born simultaneously}

The user's spatial intuition has no privileged dimension two. A higher-order
GST world birth can contain any finite number $m$ of independently
realized causal-identification channels. Crucially, the \emph{actual
cohomology and its Hodge/weight filtration} must be computed \emph{from}
the resulting geometry, not chosen by an arbitrary Hodge-to-Chow map.

Choose connected nodal curves $C_{\Gamma_i}$, $i=1,\ldots,m$,
whose normalization consists of $r_i$ copies of $\mathbb P^1$,
whose dual graph has incidence cycle rank
$\beta_i=e_i-r_i+1$. Put
\[
 W_{m,N}=\prod_{i=1}^{m} C_{\Gamma_i}\times\mathbb P^N,
 \qquad m,N\geq0.
\]
The empty product for $m=0$ is a point. This is a projective
complex scheme of dimension $m+N$; for $m>0$ and actual nodes it
is generally singular. Its entire rational graded mixed-Hodge
polynomial is computed, not postulated:
\[
\boxed{\quad
 \mathcal H_{m,N}(t,u,v,s)=
   \left(\prod_{i=1}^{m}(1+\beta_i t+r_i t^2s^2uv)\right)
   \sum_{j=0}^{N}(t^2s^2uv)^j.
\quad} \tag{5}
\]
This is an \emph{all-order causal law}: it does not impose a
maximum on the number of folded worlds or their emerged geometric
dimension. The induced Hodge data come from the realization functor,
the normalization exact sequence and K\"unneth.

\begin{theorem}[Maximum causal purity deficit: Hodge genesis for every $m$]
For each $m\geq0$, $N\geq0$ and $0\leq j\leq N$,
\[
\boxed{\quad
\operatorname{Gr}^{W}_{2j}H^{m+2j}(W_{m,N},\mathbb Q)
  \cong\mathbb Q(-j)^{\,\prod_{i=1}^{m}\beta_i}.
\quad} \tag{6}
\]
In this summand the exact causal genesis signature is
\[
(k,p,q,w)=(m+2j,j,j,2j), \qquad
\boxed{\chi=p-q=0,\quad \delta=w-k=-m.}  \tag{7}
\]
For $m>0$, the born classes are diagonal and rational but not
pure of their cohomological degree. For $m=0$ this reduces to
the genuine pure Tate class of $\mathbb P^N$.
If any $\beta_i=0$, the maximal-deficit sector vanishes, as
predicted by its \emph{actual} causal topology.
\end{theorem}
\begin{proof}
For each nodal factor, the only nonzero rational cohomology
groups are $H^0$ of degree/weight $(0,0)$,
graph $H^1$ of degree/weight $(1,0)$ and
$H^2$ of degree/weight $(2,2)$.
Thus every nodal factor contributes a nonnegative purity
deficit $k-w$ of at most one; the projective factor has
deficit zero. Degree $m+2j$ with weight $2j$ requires
total deficit exactly $m$, so \emph{every} one of the $m$
nodal factors must contribute its graph $H^1$. The
projective factor is then necessarily its degree-$2j$
Tate class. Applying mixed-Hodge K\"unneth, the unique

\section{Higher-dimensional causal boundaries actually realize Hodge worlds:

 Y_n=\bigl\{x_0x_1\cdots x_n=0\bigr\}

 \dim C_k=\binom{n+1}{k+1}.
\]

\partial_k[v_0,\ldots,v_k]
 =\sum_{i=0}^{k}(-1)^i
 [v_0,\ldots,\widehat v_i,\ldots,v_k]

\[
\boxed{\quad\operatorname{rank}\partial_k
 =\binom{n}{k},\quad1\le k\le n-1.\quad}\tag{9}
\]
Consequently its reduced rational cohomology has exactly one
nonzero group:
\[
\boxed{\quad\widetilde H^k(\partial\Delta^n,\mathbb Q)
 \cong\begin{cases}
  \mathbb Q,&k=n-1,\\0,&k\ne n-1.
 \end{cases}\quad}\tag{10}
\]
In particular, $2$-cells \emph{kill} all candidate
first-cohomology loop classes in the tetrahedral $n=3$

from the unfilled top causal sphere.
\end{theorem}
\begin{proof}

recursion, and the rank-one boundary of the omitted top cell
give the ranks $\binom nk$ (also obtained inductively from
Pascal's identity). The remaining top cycle is the signed

This proves (9)--(10).
\end{proof}

\section{The causal toric boundary determines a genuine Hodge period equation}

The all-$n$ normal-crossing-to-smooth family can be made sufficiently
explicit that the \emph{actual holomorphic middle-form period} is
computed by a universal operator.  This goes beyond a dimension
counter or a freely chosen pure-Hodge projection.

Choose the Dwork-type deformation
\[
 F_t(x)=x_0x_1\cdots x_n+
               t\sum_{i=0}^{n}x_i^{n+1},\qquad n\ge2.
\]
It is an instance of (12), and for generic sufficiently small nonzero
$t$ defines a smooth Calabi--Yau hypersurface of dimension $n-1$.
On the dense torus of $\mathbb P^n$, the defining equation factors
as the coordinate hyperplane boundary monomial times
\[
 1+tP(x),\qquad
 P(x)=\sum_{i=0}^n\frac{x_i^{n+1}}{\prod_{r=0}^n x_r}.
\]
Since $P$ is homogeneous of degree zero, it is a genuine Laurent
polynomial in any $n$ independent torus coordinates.

\begin{theorem}[Causal toric fold to holomorphic period: exact all-$n$ equation]
The normalized torus constant-term period
\[
 \Phi_n(t)=\operatorname{CT}_{(\mathbb C^\times)^n}
       \frac{1}{1+tP}
\]
is a formal power series with nonzero constant term and explicit
coefficients
\[
\boxed{\quad
\Phi_n(t)=\sum_{k\ge0}
 \frac{((n+1)k)!}{(k!)^{n+1}}\,(-t)^{(n+1)k}.
\quad} \tag{16}
\]
Let $z=(-t)^{n+1}$, $\theta=z\frac{d}{dz}$, and write
$\Phi_n(z)=\sum_{k\ge0}a_kz^k$.
Then the all-dimensional Picard--Fuchs relation is
\[
\boxed{\quad
\left[\theta^{\,n}
 -(n+1)z\prod_{r=1}^{n}\bigl((n+1)\theta+r\bigr)
 \right]\Phi_n(z)=0.
\quad} \tag{17}
\]
The torus integral of the rational differential form
$\Omega/F_t$ gives this constant-term period near the
toric boundary; $\operatorname{Res}_{X_t}(\Omega/F_t)$ is
the genuine holomorphic $(n-1,0)$ form on the smooth
geometric fiber. With the usual residue/tube comparison,
(16)--(17) govern a holomorphic middle-form period
of the emergent smooth Hodge structure.

For $n=2$ this is the cubic elliptic curve operator
$\theta^2-3z(3\theta+1)(3\theta+2)$.
For $n=3$ it is the quartic K3 operator
$\theta^3-4z(4\theta+1)(4\theta+2)(4\theta+3)$.
For $n=4$ it is the quintic threefold operator
$\theta^4-5z\prod_{r=1}^4(5\theta+r)$.
\end{theorem}
\begin{proof}
Expand the geometric series in $t$ and then expand
$P^\ell$ multinomially. Choose $k_i$ occurrences of the
$i$-th Laurent monomial, so $\sum_i k_i=\ell$.
Its exponent on $x_i$ is $(n+1)k_i-\ell$. A
simultaneous constant term in all torus coordinates
requires all $(n+1)k_i-\ell=0$.
Therefore $\ell=(n+1)k$ and
$k_0=\cdots=k_n=k$.  The unique multinomial
constant-term coefficient is
$((n+1)k)!/(k!)^{n+1}$, proving (16) without any
finite cutoff.

For $k\ge1$ the coefficient satisfies
\[
\begin{split}
k^n a_k
&=(n+1)\left(\prod_{r=1}^{n}
 ((n+1)(k-1)+r)\right)a_{k-1}.
\end{split}
\]
Indeed, the ratio of factorials
$a_k/a_{k-1}$ is
$\prod_{r=1}^{n+1}((n+1)(k-1)+r)/k^{n+1}$;
the last numerator factor is $(n+1)k$.
Comparing coefficients of $z^k$ gives (17),
including $k=0$ where both sides vanish.
The nonzero constant-term torus period is represented
analytically, for sufficiently small $t$, by integration
of the convergent geometric series against the
normalized logarithmic volume form
$\prod_{i=1}^{n} dx_i/(2\pi i x_i)$.
Poincar\'e residue supplies the holomorphic form
on $X_t$; the standard residue/tube comparison
identifies the corresponding smooth-fiber period
up to the conventional normalization.
\end{proof}

\paragraph{GST interpretation of (17).}
The signed incidence faces of the toric normal-crossing
boundary dictate the dimension of the newly born
cohomological sector.  The \emph{same} all-$n$
causal/smoothing realization determines the Laurent
polynomial, whose constant terms force the entire
holomorphic period and its order-$n$ differential
equation. The Hodge generator is therefore computed as
a geometric output of the causal-dimensional deformation,
not declared from an unrelated fixed Hodge space.
Equation (17) does not claim that its holomorphic
$(n-1,0)$ class is an algebraic $(p,p)$ cycle class;
the actual smooth algebraic diagonal correspondence
discussed above still detects its period and acts
as identity on the middle cohomology.

Exact Wolfram checks verified the factorial coefficient
recurrence for $2\le n\le8$ and $1\le k\le12$
($84$ independent integer identities). The mathematical
proof of (16)--(17) is the general multinomial argument.


\section{GST causal boundary to maximally unipotent Hodge-period monodromy:
an exact all-dimensional logarithmic tower}

The previous causal toric family already furnishes the exact
Picard--Fuchs differential operator
\[
 L_n=\theta^n-(n+1)z\prod_{r=1}^{n}((n+1)\theta+r),
 \qquad \theta=z\,d/dz,\qquad n\ge2.
\]
Here $d=n-1$ is the dimension of the smooth Calabi--Yau
hypersurface, whereas the central $n$-simplex boundary
$\partial\Delta^n$ is the actual causal dual complex of the
simple-normal-crossing world. We now extract \emph{the entire
logarithmic solution tower} rather than only its holomorphic period.

For $\rho$ near $0$ define the rational functions
\[
 a_0(\rho)=1,\qquad
 a_k(\rho)=\prod_{h=1}^{k}
 \frac{(n+1)\prod_{r=1}^{n}((n+1)(h-1+\rho)+r)}
 {(h+\rho)^n},
 \quad k\ge1,
\]
and a convergent local Frobenius family (with chosen branch of
$\log z$)
\[
 \mathcal F_n(z,\rho)=z^\rho
 \sum_{k=0}^{\infty}a_k(\rho)z^k.
\]
The coefficient ratio tends in modulus to $(n+1)^{n+1}$ as
$k\to\infty$, so these series converge locally for
$|z|<(n+1)^{-(n+1)}$, uniformly in $\rho$ on compact neighborhoods
of zero avoiding the negative integral poles.

\begin{theorem}[Causal genesis forces a maximal Picard--Fuchs
logarithmic tower in every dimension]
For the specific operator $L_n$ and the series above,
\[
 \boxed{\quad L_n\mathcal F_n(z,\rho)=\rho^n z^\rho.\quad}
 \tag{18}
\]
For each $0\le j<n$ the functions
\[
 \boxed{\quad
 \psi_j(z)=\frac1{j!}
  \left.\frac{\partial^j}{\partial\rho^j}
  \mathcal F_n(z,\rho)\right|_{\rho=0}
 \quad}\tag{19}
\]
are \emph{linearly independent actual solutions} of
$L_n\psi_j=0$. Their leading logarithmic parts are
$\psi_j(z)=(\log z)^j\Phi_n(z)/j!$ plus terms of lower
logarithmic degree; in particular $\psi_0=\Phi_n$.

Analytic continuation once anticlockwise around $z=0$
acts on this $n$-dimensional Picard--Fuchs solution local
system by
\[
\boxed{\quad
 M\psi_j=\sum_{h=0}^{j}
 \frac{(2\pi i)^{j-h}}{(j-h)!}\psi_h.
\quad}\tag{20}
\]
Consequently $M$ is unipotent with exactly one Jordan block
of size $n$, and for its logarithm
$\mathcal N=\log M$,
\[
\boxed{\quad
 \mathcal N^n=0,\qquad
 \mathcal N^{n-1}\ne0,\qquad
 \operatorname{rank}(\mathcal N^{n-1})=1.
\quad}\tag{21}
\]
Thus the length $n$ of the analytic Hodge-period degeneration
is governed by the \emph{emergent dimension} $d=n-1$.
\end{theorem}
\begin{proof}
Applying $\theta$ to $z^{k+\rho}$ multiplies it by
$k+\rho$. At exponent $z^{k+\rho}$ for $k\ge1$ the
coefficient in $L_n\mathcal F_n$ is
\[
(k+\rho)^n a_k(\rho)-
(n+1)\prod_{r=1}^{n}((n+1)(k-1+\rho)+r)
a_{k-1}(\rho),
\]
which vanishes \emph{identically} by the defining product
formula. The only remaining term is that for $k=0$,
namely $\rho^n z^\rho$, proving (18).
Differentiating this identity $j<n$ times at $\rho=0$
gives zero, proving all the differential equations.
At $\rho=0$, the product formula reduces to
$a_k(0)=((n+1)k)!/(k!)^{n+1}$ and so $\psi_0$
is the actual period series from equation (16).
Since $\mathcal F_n$ contains $z^\rho=e^{\rho\log z}$
and its analytic coefficient series has nonzero
constant term, $\psi_j$ contains a new leading
term $(\log z)^j/j!$; triangularity in powers
of $\log z$ proves independence.
A loop around zero changes $\log z$ to
$\log z+2\pi i$, and therefore multiplies
$\mathcal F_n$ by $e^{2\pi i\rho}$ while leaving
each coefficient $a_k(\rho)$ unchanged.
The Leibniz rule for the $\rho$ derivatives
gives (20) exactly. Its matrix is
$\exp(2\pi i J_n)$, where $J_n$ is the
strict one-superdiagonal Jordan nilpotent.
As $J_n^n=0$ but
$J_n^{n-1}$ has a single nonzero entry,
its logarithm
$\mathcal N=2\pi i J_n$ obeys (21).
\end{proof}

\[
\boxed{\quad
 \dim_{\mathbb Q}\widetilde H^{n-1}
    (\partial\Delta^n,\mathbb Q)

 =1,\qquad n\ge2.
\quad}\tag{22}
\]
\end{corollary}
\begin{proof}
The left side is one by the explicit signed-simplex
boundary computation (10). The right side is one

calculation (21). Both computations originate
from the \emph{same concrete toric causal family}.
\end{proof}

 \boxed{\quad\ker(M-I)=\ker\mathcal N

 \tag{24}

 M-I=\exp(\mathcal N)-I

\]
the second factor is invertible (a unipotent
