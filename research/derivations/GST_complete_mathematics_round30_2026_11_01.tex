\subsection{The rotation deck laws}

\begin{theorem}[R1; the $C_4$ obstruction law, proven]\label{thm:R1}
Let $M$ be a finitely generated free $\Z$-module carrying
$C_4=\langle\gamma\rangle$ with $\gamma^2=-\mathrm{id}_M$. Then:
$M^{C_4}=0$; the norm degenerates $N=0$; $H^2(C_4;M)=0$; and
$H^1(C_4;M)=M/(\gamma-1)M$ is $2$-elementary.
\end{theorem}

\begin{proof}
If $\gamma v=v$ then $\gamma^2v=v=-v$, so $2v=0$ and torsion-freeness gives
$v=0$. $N=(1+\gamma)(1+\gamma^2)=0$ since $\gamma^2=-1$. Hence
$H^2=M^{C_4}/NM=0$. Finally $(\gamma-1)(1+\gamma)v=(\gamma^2-1)v=-2v$, so
$2M\subseteq(\gamma-1)M$ and $H^1$ is killed by $2$.
\end{proof}

\begin{theorem}[R2; the letter-parity law, proven]\label{thm:R2}
On the letter lattice (the exterior algebra on
$\{e_0,f_0,\dots,e_3,f_3\}$) with the operator's single-factor Gaussian
rotation $\gamma_j\colon e_j\mapsto-f_j,\ f_j\mapsto e_j$ fixing all other
letters:
\begin{enumerate}
\item \textbf{Letter-parity:} $\gamma_j^2(w)=(-1)^{k_j(w)}w$ for every word
$w$ with $k_j(w)$ letters of factor $j$.
\item \textbf{All-odd sector:} if $k_j(w)=1$ for all $j$ then $\gamma_j^2=-1$
on $w$ for every $j$; R1 applies word by word.
\item \textbf{One plane:} for $\Omega=\prod_j(e_j+if_j)$,
$\gamma_j\Omega=i\Omega$, $\gamma_jW_R=-W_I$, $\gamma_jW_I=W_R$; the Weil
plane $\Z\langle W_R,W_I\rangle$ is a single $C_4$-module for every deck.
\item \textbf{All-even sector:} $\gamma_j^2=+1$ and the pair-product words
$e_jf_je_kf_k$ are deck-fixed; a trivial $C_4$-module carries
$H^2=M/4M\neq0$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) $\gamma_j^2$ negates both letters of factor $j$ and fixes the rest; the
sign of the wedge is $(-1)^{k_j(w)}$. (2) Immediate. (3)
$\gamma_j(e_j+if_j)=-f_j+ie_j=i(e_j+if_j)$; expand over the $2^4$ words and
take real and imaginary parts. (4) $k_j\in\{0,2\}$ gives $\gamma_j^2=+1$;
$\gamma_j(e_j\wedge f_j)=(-f_j)\wedge e_j=e_j\wedge f_j$; the trivial
action has $M^{C_4}=M$ and $N=4$.
\end{proof}

\begin{corollary}[Input-blindness of the deck, proven]

chain using the deck alone would descend all-odd content of every type; the
$(p,p)$-gate is not the deck --- it lives in the input step.
\end{corollary}

\begin{theorem}[R5; parity unification on the Weil rays, proven]\label{thm:R5}
On the Weil plane $M=\Z\langle W_R,W_I\rangle$ with $\gamma W_R=-W_I$,
$\gamma W_I=W_R$:
$H^1(C_4;M)=M/(\gamma-1)M\cong\Z/2$;
$m\,W_R\in\Img(\gamma-1)M\iff m$ even (likewise $W_I$);
the diagonal $W_R+W_I$ lies in $\Img(\gamma-1)$ integrally:
$(\gamma-1)(-W_R)=W_R+W_I$.
Consequently the door data --- the admitted scales $[2,4,6,8,10,12]$, the
odd refusals, and the pins $m^\ast(A,2,W_R)=m^\ast(A,2,W_I)=2$ --- is
exactly the $H^1$-law: the least scale killing the $H^1$-torsion of the ray
is the least even scale, $2$.
\end{theorem}

\begin{proof}
The matrix of $\gamma-1$ in the basis $(W_R,W_I)$ is
$\left(\begin{smallmatrix}-1&1\\-1&-1\end{smallmatrix}\right)$ with
determinant $2$: the image has index $2$ and $H^1\cong\Z/2$. Solving
$(\gamma-1)(aW_R+bW_I)=mW_R$ forces $b=-a$ and $-2a=m$: solvable exactly
for $m$ even. The $W_I$ criterion follows since $\gamma$ commutes with
$\gamma-1$ and $\gamma W_R=-W_I$. The diagonal:
$(\gamma-1)(-W_R)=W_I+W_R$.
\end{proof}

\subsection{The refined door law, full form (this round's discovery)}

\begin{theorem}[Refined door law, full form; proven here]\label{thm:refined-door}
On the one-letter product module $M=H^1(E,\Z)\otimes L=\Z\langle
e,f\rangle\otimes L$ ( $L$ torsion-free, inert) with
$\gamma(e\otimes\ell)=-f\otimes\ell$,
$\gamma(f\otimes\ell)=e\otimes\ell$, write elements as
$x=e\otimes p+f\otimes q$ with $p,q\in L$. Then:
\[
(\gamma-1)(x)=e\otimes(q-p)+f\otimes(-p-q),
\qquad
\Img(\gamma-1)=\{(r,s)\in L\oplus L:\ r+s\in2L,\ r-s\in2L\},
\]
and for every $x=e\otimes p+f\otimes q$ and every $m\ge1$:
\begin{equation}\label{eq:refined-door}
\boxed{\;m\,(e\otimes p+f\otimes q)\in\Img(\gamma-1)\iff
m(p+q)\in2L\ \text{and}\ m(p-q)\in2L.\;}
\end{equation}
In particular (the ray form, $q=0$):
$m\,(e\otimes\ell)\in\Img(\gamma-1)\iff m\,\ell\in2L$.
\end{theorem}

\begin{proof}
The first display is the substitution
$\gamma(e\otimes p)=-f\otimes p$, $\gamma(f\otimes q)=e\otimes q$. For the
image characterization: $(r,s)=(q-p,-p-q)$ has
$r+s=-2p\in2L$ and $r-s=2q\in2L$; conversely, given $r+s=2u$ and
$r-s=2v$ (so $r=u+v$, $s=u-v$), take $p=-u$, $q=v$: then
$q-p=u+v=r$ and $-p-q=u-v=s$. For \eqref{eq:refined-door}:
$m\,x=(m p, m q)$ lies in $\Img(\gamma-1)$ iff $mp+mq=m(p+q)\in2L$ and
$mp-mq=m(p-q)\in2L$. The ray form is $q=0$.
\end{proof}

\begin{theorem}[The conjugation theorem; proven here]\label{thm:hodge-typing}
Let $\ell\in H^{2p-1}(Y,\Z)$ be a nonzero class of the free part
(equivalently: an integral class with nonzero complexification). Then the
complexified class $\ell_\C\in H^{2p-1}(Y,\C)$ is fixed by conjugation,
and therefore is \textbf{never} purely of type $(p-1,p)$ ( nor of type
$(p,p-1)$ ): a conjugation-fixed class of pure type $(p-1,p)$ lies in
$H^{p-1,p}\cap H^{p,p-1}=0$.
Consequently the \emph{pure-ray} input set of the product world
\emph{is empty}: there is no nonzero $\ell\in L_Y$ with $e\otimes\ell$
purely of type $(p,p)$ on $E\times Y$.
\end{theorem}

\begin{proof}
Conjugation acts on $H^{2p-1}(Y,\C)$ as the conjugate-linear involution
exchanging $H^{p-1,p}$ and $H^{p,p-1}$ while fixing the image of
$H^{2p-1}(Y,\Z)$ ( $\bar\ell=\ell$ for integral $\ell$ ). If $\ell_\C\in
H^{p-1,p}$ purely, then $\ell_\C=\bar{\ell}_\C\in H^{p,p-1}$, so
$\ell_\C\in H^{p-1,p}\cap H^{p,p-1}=0$.
For the ray: $e$ decomposes as $e=\tfrac12(\varepsilon+\eta)$ with
$\varepsilon=e+if$ spanning $H^{1,0}(E)$ ( the $+i$-eigenline of
$[\,i\,]^\ast$ ) and $\eta=e-if$ spanning $H^{0,1}(E)$; hence
$e\otimes\ell=\tfrac12\varepsilon\otimes\ell+\tfrac12\eta\otimes\ell$,
which is purely $(p,p)$ iff $\ell$ is purely $(p-1,p)$ and purely
$(p,p-1)$ simultaneously --- impossible for nonzero integral $\ell$.
\end{proof}

\begin{theorem}[The balanced inputs; proven here]\label{thm:balanced}
The $(p,p)$ Hodge inputs of the product world in the one-letter sector
are exactly the \textbf{balanced} classes
\[
x=e\otimes p+f\otimes q,\qquad p,q\in L_Y\ \text{integral},
\]
whose complex combination $\,p-iq\,$ is purely of type $(p-1,p)$ on $Y$
(equivalently: $p+iq$ purely $(p,p-1)$ --- automatic, being the
conjugate). For these, the full-form door law
\eqref{eq:refined-door} applies verbatim: $m\,x$ is admitted iff
$m(p+q)\in2L_Y$ and $m(p-q)\in2L_Y$.
\end{theorem}

\begin{proof}
With $e=\tfrac12(\varepsilon+\eta)$, $f=\tfrac{-i}{2}(\varepsilon-\eta)$:
\[
x=\tfrac12\,\varepsilon\otimes(p-iq)\;+\;\tfrac12\,\eta\otimes(p+iq).
\]
The $\varepsilon$-term is $(1,0)\otimes(\text{Y-part})$, of type $(p,p)$
iff the Y-part is $(p-1,p)$; the $\eta$-term is then automatically
$(p,p-1)\otimes(0,1)=(p,p)$, being the conjugate term ( $p,q$ real ).
For the exemplar: $Y=E'$, $p=e'$, $q=f'$: $p-iq=e'-if'=\eta'$ spans
$H^{0,1}(E')$ --- purely $(0,1)$, the $p=1$ case --- so
$x=e\otimes e'+f\otimes f'$ is purely $(1,1)$; and
$x=\tfrac12(\varepsilon\otimes\eta'+\eta\otimes\varepsilon')$ is
integral ( the Poincaré class of $E\times E'$ ). Its door: $p+q=e'+f'$,
$p-q=e'-f'$, both primitive --- admitted iff $m$ even, refused at $m=1$.
\end{proof}
