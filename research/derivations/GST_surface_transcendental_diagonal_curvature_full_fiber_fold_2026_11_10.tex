\section{Transcendental diagonal as an independently algebraic source}
Let $S$ be a smooth connected projective complex surface with $H^1(S,\Q)=0$ and Picard number $\rho\ge1$. Put $b_2=\dim_\Q H^2(S,\Q)$ and $t=b_2-\rho>0$. Choose algebraic divisors $D_1,\ldots,D_\rho$ spanning the rational Neron--Severi group, and write $G_{ij}=\int_SD_iD_j$. The rational matrix $G$ is invertible by the Hodge index theorem. Set $X=S\times S$, an \emph{actual smooth projective original fourfold}; in $\CH^2(X)_\Q$ set
\[
a=[pt]\times S,\quad b=S\times[pt],\quad
d_{ij}=\operatorname{pr}_1^*D_i\operatorname{pr}_2^*D_j,\quad
U=\sum_{i,j}(G^{-1})_{ij}d_{ij},
\]
\[
\boxed{\Xi=[\Delta_S]-a-b-U.}
\]
Each term is an actual algebraic codimension-two cycle. The diagonal acts as identity on $H^*(S)$, $a+b$ projects to $H^0\oplus H^4$, and $U$ projects onto the rational Neron--Severi part of $H^2$. Thus the cohomological correspondence $\cl(\Xi)$ is identity on the transcendental complement $T_S$ and zero on the other summands.

\begin{proposition}[Original-X primitive geometric source]
\[
 \int_X\Xi a=\int_X\Xi b=\int_X\Xi d_{ij}=0,
 \qquad\boxed{\int_X\Xi^2=t=b_2-\rho>0.}
\]
For an ample $H\in\CH^1(S)$, $d=\int_SH^2>0$, $c=H_1H_2$ and $P=c-d(a+b)$, both $\cl(P),\cl(\Xi)$ are primitive with respect to $H_1+H_2$, and
\[
\boxed{\left(\int_X Z_iZ_j\right)_{Z=(P,\Xi)}
=\begin{pmatrix}3d^2&0\\0&t\end{pmatrix}.}
\]
In particular $\Xi$ is not cohomologically a rational combination of divisor products on $X$.
\end{proposition}
\begin{proof}
For divisor products $\int_Xd_{ij}d_{k\ell}=G_{ik}G_{j\ell}$ and $\int_X[\Delta_S]d_{ij}=G_{ij}$.
The diagonal self-intersection is $\int_X[\Delta_S]^2=\chi_{\rm top}(S)=2+b_2$ because $H^1=H^3=0$.
Directly $\int_XU^2=\int_X[\Delta_S]U=\rho$, $(a+b)^2=2$, and $\int_X[\Delta_S](a+b)=2$, with all other cross terms zero. Therefore
$\int_X\Xi^2=(2+b_2)-(2+\rho)=t$, and $\Xi$ is orthogonal to $a,b,d_{ij}$.
Because $T_S$ is orthogonal to any ample $H$, cup product by $H$ sends $T_S$ to zero in $H^4(S)$. Hence $(H_1+H_2)\cl(\Xi)=0$.
The identities $a^2=b^2=ac=bc=0$, $ab=1$, $c^2=d^2$, and $(H_1+H_2)c=d(H_1+H_2)(a+b)$ show $P$ primitive with square $3d^2$. Finally $\Xi$ is orthogonal to $P$. Since the divisor-product span is generated by $a,b,d_{ij}$ for $H^1(S)=0$, its orthogonal nonzero-square class $\cl(\Xi)$ lies outside that span.
\end{proof}

\section{Actual native GST curvature and detector}
Define genuine rational Chow correspondences in $\CH^4(X\times X)_\Q$:
\[
 A=\operatorname{pr}_1^*(P/(3d^2))\operatorname{pr}_2^*P,\qquad
 B=\operatorname{pr}_1^*(P/(3d^2))\operatorname{pr}_2^*\Xi.
\]
The canonical push-pull action gives $A_*P=P$, $A_*\Xi=0$, $B_*P=\Xi$, $B_*\Xi=0$, so
\[
 \boxed{[A_*,B_*]P=-\Xi.}
\]
The fixed geometric polarization readout $\ell(x)=\int_Xx\,\Xi/t$ vanishes on all divisor products and has $\ell([A_*,B_*]P)=-1$. This implements the existing GST native commutator-birth theorem with \emph{new independent geometric input supplied by the diagonal}, not an arbitrary Chow preimage of a Hodge target.

\section{Full Hodge-fiber Chow cyclicity}
\begin{theorem}[Uniform single-fold family]
Assume additionally $\End_{\mathrm{HS}}(T_S)=\Q\cdot\mathrm{id}$. Then for every $p=0,1,2,3,4$ the actual original $X=S\times S$ admits a rational algebraic seed $Z_{0,p}$, a genuine unrestricted rational Chow correspondence $\Gamma_p\in\CH^4(X\times X)_\Q$, and an intersection detector $\ell_p$ whose full rational $(p,p)$ Hodge moment matrix is nonsingular. Every rational Hodge class on this $X$ is therefore represented by an explicit finite rational sum of actual original-X Chow cycles, without assuming its prior algebraicity.
\end{theorem}
\begin{proof}
The rational weight-four Hodge classes in $T_S\otimes T_S$ identify by polarization with $\End_{\rm HS}(T_S)$; the mixed $T_S\otimes\mathrm{NS}$ summands have no rational Hodge classes. The hypothesis makes $\cl(\Xi)$ the single transcendental diagonal generator. Consequently the entire middle rational Hodge fiber has the explicit algebraic basis
\[
 v=(a,b,(d_{ij})_{1\le i,j\le\rho},\Xi),
 \qquad n=\rho^2+3.
\]
Its genuine intersection Gram matrix is
\[
 \mathcal G=\begin{pmatrix}0&1\\1&0\end{pmatrix}
 \oplus(G\otimes G)\oplus(t),
 \qquad \boxed{\det\mathcal G=-t(\det G)^{2\rho}\ne0}.
\]
Let $\check v_i=\sum_k(\mathcal G^{-1})_{ik}v_k$ be the corresponding \emph{actual algebraic} dual cycles. Then $\int_X v_j\check v_i=\delta_{ij}$ and
\[
 \Gamma_2=\sum_{j=0}^{n-1}
 \operatorname{pr}_1^*\check v_j\operatorname{pr}_2^*v_{j+1\bmod n}
 \quad\in\CH^4(X\times X)_\Q
\]
acts by a cyclic permutation of the $n$ genuine middle Hodge basis cycles.
With $Z_{0,2}=v_0=a$ and $\ell_2(x)=\int_Xx\check v_0$,
\[
 M_{ij}=\ell_2((\Gamma_2)_*^{i+j}\cl(v_0))
 ={\bf1}_{i+j\equiv0\pmod n},\quad
 \boxed{\det M=(-1)^{(n-1)(n-2)/2}\ne0}.
\]
This is a permutation matrix with index zero fixed and the remaining $n-1$ indices reversed. The exact moment-only inverse construction of the existing GST single-fold theorem now yields a native algebraic representative for every prescribed middle Hodge class.
For $p=1$, $H^1(S)=0$ and Lefschetz $(1,1)$ give the actual divisor basis $\mathrm{NS}_\Q\oplus\mathrm{NS}_\Q$; for $p=3$ its algebraic complementary-degree pairing gives actual point-divisor cycles; $p=0,4$ have fundamental/point cycles. In each case the complementary degree geometric intersection matrix is invertible and the same algebraic cyclic-shift Chow correspondence produces a nonzero Hankel determinant. This proves all weights.
\end{proof}

\section{A hard boundary for the universal GST attempt}
If $e=\dim_\Q\End_{\rm HS}(T_S)>1$, the rational middle Hodge fiber has dimension $\rho^2+2+e$, while the above cycles have only $\rho^2+3$ independent classes. Every polynomial word in the diagonal and the Neron--Severi/degree projectors acts as a scalar on $T_S$; pre/post-composition of the transcendental diagonal by those words cannot generate a non-scalar Hodge endomorphism. Thus the restricted diagonal-only constructor fails when $e>1$. This does not refute a richer GST birth algorithm using genuinely new algebraic CM correspondences. For even $\rho$, $n=\rho^2+3$ is odd, so an operator $J^2=-I$ cannot cover the whole middle fiber; CM must live on compatible summands.
