\section{Unconditional original-X geometric source birth}
\begin{theorem}[Genuine alpha-dependent signed real degeneracy source]
For ANY rational even-degree alpha in H^(2p)(X,Q),
there are a positive integer m, an ample-class
rational coefficient lambda, and actual finite-rank
COMPLEX TOPOLOGICAL bundles E_+ and E_- on original X,
whose entire Chern characters have precisely
\[
 \operatorname{ch}(E_+)=r_++m(\alpha+\lambda h^p),
 \qquad
 \operatorname{ch}(E_-)=r_-+m\lambda h^p.       \tag{1}
\]
Let c=(-1)^(p-1)(p-1)!m.
There exist generic smooth section tuples
of cardinality rank(E_\pm)-p+1 whose rank-deficiency
loci D_+ and D_- are ACTUAL compact oriented
Whitney-stratified integral pseudocycles on X,
of main real codimension 2p, with
\[
 \mathrm{PD}_X[D_+]=c(\alpha+\lambda h^p),
 \qquad
 \mathrm{PD}_X[D_-]=c\lambda h^p.               \tag{2}
\]
Consequently the target-SELECTED signed rational
GEOMETRIC cycle
\[
 T_\alpha=\frac{(-1)^{p-1}}{m(p-1)!}
                   ([D_+]-[D_-])
\quad\text{satisfies}\quad
 \boxed{\mathrm{PD}_X T_\alpha=\alpha.}         \tag{3}
\]
No original-X algebraic cycle, target Chow preimage,
or target-dependent full-rank algebraic source is
used in the construction. The actual D_\pm are
smooth-real stratified spaces, not assumed
holomorphic/algebraic varieties.
\end{theorem}
\begin{proof}
Rational topological K-theory identifies
K^0(X) tensor Q with even rational cohomology.
Represent the two indicated rational pure graded
classes by virtual K-classes; clear a common
positive integer denominator m, realize each
integral virtual K-class by differences of honest
complex bundles, add a finite topological
complement to its negative part and stabilize ranks.
The positive Chern character components in all
degrees other than p vanish. The Newton
identity
\[
 \log c(E,t)=\sum_{j\ge1}(-1)^{j-1}
                 (j-1)!\operatorname{ch}_j(E)t^j
\]
therefore gives precisely equation (2)
for the respective pth Chern classes.
For s=rank(E)-p+1, the main rank-(s-1)
determinantal stratum of Hom(C^s,C^rank E)
has complex codimension p, while the
next rank-(s-2) stratum has complex
codimension 2p+2.
A generic smooth section tuple is transverse
to every rank stratum. The resulting
stratified degeneracy set is a compact
oriented integral pseudocycle; its main
stratum has real codimension 2p and the
higher strata have sufficiently high real
codimension to define a closed integral
homology class. The topological
Thom--Porteous formula identifies its
Poincare dual with c_p(E). Subtracting
the two EXACT rational Chern class
identities proves (3).
\end{proof}

\begin{proof}

\end{proof}

\section{Necessary signed balancing, in every dimension}
For an actual Kähler form omega representing h,
any nonempty effective analytic codimension-p

$\int_C\omega^{d-p}>0$.
But any nonzero primitive r obeys
$\int_Xr h^{d-p}=0$ since
h^(d-2p+1)r=0. Thus a SINGLE effective
analytic cycle cannot represent r even when
r is algebraic as a SIGNED rational combination.
One must use signed sources.
Choose lambda rational of sufficiently large

 c\lambda\int_Xh^d>0,\qquad
 c\left(\int_X\alpha h^{d-p}
              +\lambda\int_Xh^d\right)>0.         \tag{5}
\]
This removes the trivial single-cycle
sign obstruction but does not assert
that the real cycles become analytic.

\section{A rigorous geometric return from quantitative estimates}
Let k=d-p and phi=omega^k/k!,
the Wirtinger calibration. For any
closed integral 2k-current T define
its actual causal-complexity defect
\[
 \Delta_\omega(T)={\bf M}(T)-T(\phi)\ge0,        \tag{6}
\]
where M is geometric mass and positivity follows

\begin{theorem}[Genuine ORIGINAL-X Chow cycle from bounded-mass asymptotically calibrated signed sources]
Suppose that for the SPECIFIC alpha-dependent
bundles E_+,E_- above there are sequences of

D_{\pm,j}, each in its fixed topological

\[
 \boxed{\sup_j {\bf M}(D_{\pm,j})<\infty,\qquad
        \Delta_\omega(D_{\pm,j})\to0.}          \tag{7}
\]
Then there are ACTUAL positive algebraic
cycles C_+,C_- on ORIGINAL X with
$\mathrm{PD}_X[C_\pm]=c_p(E_\pm)$
and the explicit true rational Chow cycle
\[
 \boxed{
 Z_\alpha=\frac{(-1)^{p-1}}{m(p-1)!}
       ([C_+]_{\rm Chow}-[C_-]_{\rm Chow})
 \in CH^p(X)_{\mathbb Q},\qquad
 cl_X^{can}(Z_\alpha)=\alpha.
 }                                                       \tag{8}
\]
\end{theorem}
\begin{proof}
Compactness of X, uniform mass and zero
current boundaries give convergent flat
subsequences by Federer--Fleming:
D_{\pm,j} tends to closed integral
currents C_\pm in the same homology class.
The smooth closed calibration phi evaluates
to a fixed number on each homology class.
Lower semicontinuity of mass and (7) imply
M(C_\pm)<=liminf M(D_{\pm,j})
 =lim D_{\pm,j}(phi)=C_\pm(phi)<=M(C_\pm).
Thus C_\pm are positive Kähler-calibrated
integral currents. The standard structure
theorem for positive closed integral
complex currents identifies them with
positive holomorphic chains. They are
compact analytic cycles on the projective X
and hence actual algebraic cycles by Chow's
theorem. The class comparison, (2) and (3)
yield the exact original-X Betti formula.
\end{proof}

\[
 \boxed{
 |\psi_x(\xi)|\le
 C_k\|\psi_x\|_{\rm comass}
  \sqrt{1-\varphi_k(\xi)},

 }                                                       \tag{10}
\]
Indeed the oriented real Grassmannian

of the nonnegative Wirtinger defect,

The evaluation of an off-bidegree

normal angle. The ratio in (10)
is therefore bounded near the
zero set, and bounded away from

finite-dimensional Kähler
