\section{The letter model, stated once}

Fix the committed letter model (Rounds 25--34; restated for

\begin{definition}[The CM elliptic letter model]\label{def:letters}
Let $E_1,\dots,E_n$ be elliptic curves, grouped: each $E_i$ either has
CM by an imaginary quadratic field $K_{\rho(i)}$ (the \emph{CM
factors}), or is non-CM; non-CM factors are grouped into
\emph{GSp-blocks} $B$ (a block is a maximal set of pairwise isogenous
non-CM factors; pairwise non-isogenous non-CM factors lie in distinct
blocks). Write $V_i=H^1(E_i,\Q)$. A \emph{word of degree $2p$} is an
assignment of $2p$ tensor letters to the factors (letter $\ell$ sits on
factor $i(\ell)$; $m_i$ letters on factor $i$, $\sum_i m_i=2p$); the
word indexes the K\"unneth piece
$\bigotimes_i \mathrm{Sym}^{m_i}$-sectors of
$H^{2p}(E_1\times\dots\times E_n,\Q)$'s $H^1$-part (the $H^2$-letters
--- point classes --- attach trivially and are carried by the divisor
$\theta$-supply; see Lemma \ref{lem:pairing}).
For a CM factor $i$ with field $K=K_{\rho(i)}$, the Deligne torus acts on
$V_i\otimes\C=L_{\sigma}\oplus L_{\bar\sigma}$ through the two embeddings
$\sigma,\bar\sigma$ of $K$; a letter $\ell$ on factor $i$ carries
\emph{character} $\sigma$ or $\bar\sigma$ (its slot's $L$-line). For the
torus $T_K=\mathrm{Res}_{K/\Q}\mathbb{G}_m$ (the Mumford--Tate group of
a CM elliptic curve, and of any product of CM elliptic curves sharing
the field $K$), a rational tensor in the word is a Hodge class of degree
$2p$ exactly when its character is trivial after the twist by
$\Q(p)$ --- equivalently:
\begin{center}
\emph{per CM field $K$, the number of $\sigma$-letters equals the number
of $\bar\sigma$-letters.}
\end{center}
This is the operator's letter-parity law in the CM case (the parity is a
character count --- type-blind, deck-blind). Non-CM blocks carry no
$\sigma/\bar\sigma$ split; their Hodge classes are the
$\GSp$-invariants (Lemma \ref{lem:generic}).
\end{definition}

\begin{lemma}[The character criterion]\label{lem:character}
In the letter model of Definition \ref{def:letters}, the space of degree-$2p$
Hodge classes on the $H^1$-part of
$X=E_1\times\dots\times E_n$ has $\Q$-dimension
\[
\dim \;=\; \prod_{K}\binom{m_K}{m_K/2}\;\cdot\;\prod_{B} c(m_B),
\]
where $m_K$ is the number of letters on factors of CM field $K$
(the factor vanishes unless $m_K$ is even), and $c(m_B)$ is the dimension
of the degree-$m_B$ piece of the $\GSp$-invariant algebra of block $B$
($c(2r)=\frac{1}{r+1}\binom{2r}{r}$, the Catalan number, vanishing on
odd $m_B$; Lemma \ref{lem:generic}).
\PROVEN{} \MINE{} (certificate V1: the dimension computed two independent
ways --- direct balanced-assignment enumeration and the constant term of
the exact Laurent character polynomial
$\prod_\ell(x_{K(\ell)}+x_{K(\ell)}^{-1})$; V3: the Catalan counts by
hook length and recurrence, and the generic span rank by exact
evaluation on the standard module.)
\end{lemma}

\begin{proof}
The Mumford--Tate group of a product is the product of the per-field
tori and the per-block $\GSp$-factors acting jointly on the shared
letters; a tensor is Hodge iff it is $\MT$-invariant after the twist,
and for the torus part this is exactly the per-field character balance
(the $\Q(p)$-twist contributes the norm character
$\sigma\bar\sigma$ once per letter-pair, cancelling the twist). The
$\C$-dimension of the balanced space per field $K$ with $m_K$ letters is
$\binom{m_K}{m_K/2}$ (choose which slots carry $\sigma$); the invariant
space is defined over $\Q$ (the character condition is
Galois-stable: conjugation swaps $\sigma\leftrightarrow\bar\sigma$
globally and preserves the counts), so the $\Q$-dimension equals the
$\C$-dimension. The $\GSp$-blocks are computed in Lemma \ref{lem:generic}.
\end{proof}

\begin{lemma}[The generic block; the Catalan count]\label{lem:generic}
Let $B$ be a GSp-block with $m_B=2r$ letters on the standard $2$-dimensional
module $V$ (the non-CM elliptic $H^1$, with $\MT$ the symplectic
similitude group of the polarization). The degree-$2r$ Hodge classes on
the block's $H^1$-part form the invariant space of
$\otimes^{2r}V\otimes\det^{-r}$, of dimension the Catalan number
$\frac{1}{r+1}\binom{2r}{r}$, and it is spanned by the products of the
pairing classes (the graphs $\Gamma_{\pm 1}$ of $\pm1$ and the
polarization/point pairs on the factors).
\PROVEN{} \MINE{} (certificate V3: the Catalan number by hook-length
formula and by recurrence; the pairing-span rank by exact RREF over $\Q$
of the matching products evaluated on the standard module:
rank $=$ Catalan at $p=2,3,4$; the raw matching count $(2r-1)!!$
overshoots the rank --- the syzygies are present and computed.)
\end{lemma}

\begin{proof}
By Schur--Weyl for $\mathrm{GL}_2$,
$\otimes^{2r}V\cong\bigoplus_\lambda S_\lambda(V)\otimes[\lambda]$ over
partitions $\lambda$ of $2r$ with at most $2$ rows; the invariant line
$\det^{r}$ is $S_{(r,r)}(V)$, whose multiplicity is the number of
standard tableaux of shape $(r,r)$, i.e.\ the Catalan number (hook
length). The first fundamental theorem for the symplectic group: the
invariant algebra is generated by the degree-$2$ pairings; the products
of pairings span (the certificate computes the span dimension exactly:

(a divisor on $E_i\times E_j$), the polarization $\theta_i$ and the
point class on each factor.
\end{proof}

\section{Theorem S11: the CM pairing-generation theorem}

\begin{lemma}[The pairing space and the descent]\label{lem:pairing}
Let $E_i,E_j$ be CM elliptic curves with the same field $K$, letters
$a$ (on $i$) and $b$ (on $j$). The space of $(1,1)$-classes in
$V_i\otimes V_j$ is $2$-dimensional over $\Q$, spanned by the classes of
the graphs $\Gamma_1$ and $\Gamma_\alpha$ of $1$ and a generator
$\alpha$ of $K$ over $\Q$; on the $\C$-assignment basis
$\{e_{\sigma_a\bar\sigma_b},e_{\bar\sigma_a\sigma_b}\}$ the two graphs are
the conjugation-stable combinations
$\Gamma_1\propto e_{\sigma_a\bar\sigma_b}+e_{\bar\sigma_a\sigma_b}$ and
$\Gamma_\alpha\propto i\,(e_{\sigma_a\bar\sigma_b}-e_{\bar\sigma_a\sigma_b})$.
If $K_i\neq K_j$ (different fields), or if one factor is non-CM and the
other is not in its isogeny class, the pairing space of $(1,1)$-classes
is $\{0\}$ --- there is no Hodge pairing across the block boundary, and
the $\Hom$-supply is correspondingly zero. If $i=j$, the pairing space
is the $2$-dimensional space spanned by the polarization $\theta_i$ and
the point class (both divisor classes on $E_i$ itself).
\PROVEN{} \MINE{} (certificate V2: the two graph classes as exact
Gaussian rationals, $\Gamma_\alpha^2=-1$, the conjugation stability
$gconj(\Gamma_\alpha)=-\Gamma_\alpha$; V4/V6-H2: the cross-field supply
is zero and the sabotage filter fires.)
\end{lemma}

\begin{proof}
The $(1,1)$-part of $V_i\otimes V_j\otimes\C$ is
$(L_\sigma\otimes L_{\bar\sigma})\oplus(L_{\bar\sigma}\otimes L_\sigma)$,
$2$-dimensional. The graph of $f\in\Hom^0(E_i,E_j)=K$ (same field) is
the divisor $\{(x,fx)\}\subset E_i\times E_j$; its class
$[\Gamma_f]\in V_i\otimes V_j\otimes\Q(1)$ is the tensor of $f^*$, which
on the eigenlines is $\mathrm{diag}(\sigma(f),\bar\sigma(f))$ --- a
Hodge class (type $(1,1)$: each eigen-component pairs $\sigma$ with
$\bar\sigma$), $\Q$-rational (the graph is an integral subvariety), and
equal to
$\sigma(f)\,e_{\sigma_a\bar\sigma_b}+\bar\sigma(f)\,e_{\bar\sigma_a\sigma_b}$
--- the conjugation-stable combinations exactly as claimed, since
$\bar{\sigma(f)}=\bar\sigma(f)$. Taking $f\in\{1,\alpha\}$ with
$\sigma(\alpha)=i$ up to normalization gives the two displayed spanning
classes; they are $\Q$-independent since
$\{1,i\}$ spans $K\otimes_\Q\C$-stably. Conversely every
$\Q$-rational $(1,1)$-class in $V_i\otimes V_j$ is
conjugation-stable, hence of the form
$c\,e_{\sigma_a\bar\sigma_b}+\bar c\,e_{\bar\sigma_a\sigma_b}$, hence ---
$K\otimes\Q\to\C$ hitting all such $c$ --- a $\Q$-combination of the two
graphs: the graphs span. For different fields, no character balance
exists across $K_i\neq K_j$ (a $\sigma_{K_i}$-letter cannot pair with a
$\sigma_{K_j}$-letter: the torus
$T_{K_i}\times T_{K_j}$ acts with four distinct characters on
$(V_i\otimes V_j)\otimes\C$ and the trivial-after-twist character forces
the mixed components to vanish); $\Hom^0(E_i,E_j)=0$ confirms the supply
side. The same argument covers the non-CM/block-boundary case
($\Hom^0=0$ across non-isogenous factors; the character structures
differ).
\end{proof}

\begin{theorem}[The CM Pairing-Generation Theorem]\label{thm:s11}
Let $X=E_1\times\dots\times E_n$ be a product of elliptic curves
(any mix of CM fields and non-CM blocks). Then every rational Hodge
class on $X$, of any degree $2p$, is a $\Q$-combination of products of
divisor classes on $X$. In particular the rational Hodge conjecture
holds on $X$ unconditionally, at every degree.
\PROVEN{}
\end{theorem}

\begin{proof}
By the K\"unneth decomposition, a degree-$2p$ class is a sum of words
(\S2); the $H^2$-letters (point classes on single factors) are divisor
classes and can be pulled out; it suffices to treat each pure
$H^1$-word $w$ separately. Let $w$ have $m_K$ letters on each CM field
$K$ and $m_B$ on each block $B$.

\emph{Case CM.} If some $m_K$ is odd, the word carries no Hodge classes
at all (Lemma \ref{lem:character}: the balanced count vanishes) ---
nothing to prove. Otherwise, let $S$ be the set of balanced
$\sigma/\bar\sigma$-assignments on $w$'s letters;
$|S|=\prod_K\binom{m_K}{m_K/2}$, and this is the word's Hodge space
dimension (Lemma \ref{lem:character}). The supply: for each assignment
$A\in S$ and each CM field $K$, match $A$'s $\sigma$-letters on field
$K$ with $A$'s $\bar\sigma$-letters on field $K$ (a perfect matching
exists: equal counts), pairing letters of the same field pairwise (any
two letters on factors of field $K$, same or different factors). Each
pair $(a,b)$ carries the two graph classes of Lemma \ref{lem:pairing}
(one letter-pair per pair of slots, all divisor classes on the
corresponding $E\times E'$ or on $E$ itself for same-factor pairs).
The product of the $p$ pair classes is a degree-$2p$ class, a product of
divisors, supported exactly on the assignment $A$ in the sense that its
$(A')$-component is $\delta_{A,A'}$ up to the per-pair graph
coefficients --- concretely: the product with all pairs in the
$\Gamma_1$-component is the basis tensor $e_A$ itself (each
$\Gamma_1$-pair forces its two letters to the $(\sigma,\bar\sigma)$ or
$(\bar\sigma,\sigma)$ orientation chosen by the matching, which is
$A$'s). Hence the $\{e_A\}_{A\in S}$ are supply products, and the Hodge
space --- spanned by $\{e_A\}$ over $\C$, and by its
Galois-descent (conjugation maps $e_A\leftrightarrow e_{\bar A}$,
$\bar A$ the globally flipped assignment, also balanced) over $\Q$ ---
is spanned over $\Q$ by the supply products together with their
conjugates, which are again supply products (the conjugate of a
$\Gamma_1$-product is a $\Gamma_1$-product; the conjugate of a
$\Gamma_\alpha$-pair is its negative, Lemma \ref{lem:pairing}). Every
rational Hodge class on $w$ is therefore a $\Q$-combination of products
of divisor classes. The certificate verifies the span exactly
(V2: the Gaussian-rational RREF rank of the full matching-product matrix
equals $|S|$ at every committed shape, including the mixed-field
shapes).

\emph{Case GSp-block.} Lemma \ref{lem:generic}: the block's Hodge space
is Catalan-dimensional and spanned by pairing products --- divisors on
the block's factors.

\emph{Mixed words.} The word's Hodge space is the tensor product of the
per-field and per-block invariant spaces (the $\MT$ is the product), and
the supply products factor accordingly; both factors are divisor-product
spans, hence so is the product.

Combining the words and re-attaching the point classes:
every rational Hodge class on $X$ is a $\Q$-combination of products of
divisor classes. Since divisor classes are algebraic
(Lefschetz $(1,1)$; the graphs are subvarieties), products of algebraic
classes are algebraic, and $\Q$-combinations of algebraic classes are
algebraic: the rational Hodge conjecture holds on $X$, unconditionally,
at every degree.
\end{proof}
