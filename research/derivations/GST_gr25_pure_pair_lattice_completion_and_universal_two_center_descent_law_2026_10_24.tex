\section{The universal two-center descent law (all $p$)}

The two lemmas and the theorem below are proved for arbitrary smooth
complex projective $X$, $1\le p\le\dim X$, and rational
$\alpha\in H^{2p}(X,\mathbb Q)$; they use only the splitting
principle, Newton's identities, and Thom--Porteous. This is the
extracted descent law, with all constants pinned.

\begin{lemma}[Newton conductor]
\label{lem:newton}
Let $V$ be a virtual $K$-class ($V=[E_+]-[E_-]$, exact bundles of
equal rank $r$) with
\[
\ch_j(V)=0\;\;(1\le j<p),\qquad \ch_p(V)=m\alpha .
\]
Then the Chern classes of the virtual difference satisfy
\[
c_j(V)=0\;\;(1\le j<p),\qquad
\boxed{\;c_p(V)=(-1)^{p-1}(p-1)!\,m\,\alpha\;}
\]
where $c(V)$ is defined by the total Chern class of the virtual
difference, $c(V):=c(E_+)/c(E_-)$.
\end{lemma}

\begin{proof}
Write the power sums of the virtual Chern roots as
$p_k(V):=\sum_i x_i^k-\sum_j y_j^k$ with $\{x_i\},\{y_j\}$ the root
multisets; then $p_k(V)=k!\,\ch_k(V)$, so $p_j(V)=0$ for $j<p$ and
$p_p(V)=p!\,m\alpha$. Newton's identities
\[
p_k-e_1p_{k-1}+e_2p_{k-2}-\dots+(-1)^{k-1}e_{k-1}p_1
+(-1)^k k\,e_k=0
\]
express $e_k=c_k(V)$ as $(-1)^{k+1}p_k/k$ plus a polynomial with zero
constant term in $e_1,\dots,e_{k-1}$, and dually express $p_k$ as
$(-1)^{k-1}ke_k$ plus a polynomial with zero constant term in
$p_1,\dots,p_{k-1}$. Induction on $k<p$: $p_k=0$ for $k<p$ and the
zero-constant-term structure give $e_k=0$ for $k<p$. At $k=p$ all
$e_j$ ($j<p$) vanish, so $p_p=(-1)^{p-1}p\,e_p$, i.e.\
$c_p(V)=e_p=(-1)^{p-1}p!\,m\alpha/p=(-1)^{p-1}(p-1)!\,m\alpha$.
\end{proof}

\begin{lemma}[Twist cancellation for pure pairs]
\label{lem:twist}
Let $E_+,E_-$ be exact bundles of equal rank $r$ with
$c_j(E_+)=c_j(E_-)$ for $1\le j<p$. Then for EVERY line bundle $L$
with $\ell:=c_1(L)$,
\[
c_p(E_+\tensor L)-c_p(E_-\tensor L)=c_p(E_+)-c_p(E_-).
\]
\end{lemma}

\begin{proof}
By the splitting principle, with $x_1,\dots,x_r$ the Chern roots of
$E_\pm$,
\[
c_p(E\tensor L)=e_p(x_1+\ell,\dots,x_r+\ell)
=\sum_{t=0}^{p}\binom{r-t}{p-t}\,c_t(E)\,\ell^{\,p-t},
\]
since a term choosing the $x$-factor from a $t$-subset $T$ arises from
the $\binom{r-t}{p-t}$ supersets $S\supset T$, $|S|=p$. Subtracting
the two expansions, all $t<p$ terms cancel by hypothesis, and the
$t=p$ term carries coefficient $\binom{r-p}{0}=1$.
\end{proof}

\begin{theorem}[Universal two-center descent law]
\label{th:law}
Let $X$ be smooth complex projective, $1\le p\le\dim X$,
$\alpha\in H^{2p}(X,\mathbb Q)$ rational, $m\ne0$ an integer. Suppose
given a \emph{pure pair at scale $m$}: exact holomorphic bundles
$E_\pm$ of equal rank $r$ with
\begin{equation}
\ch(E_+)-\ch(E_-)=m\,\alpha
\quad\text{with all components off grade $p$ vanishing.}
\label{eq:pair}
\end{equation}
Let $L$ be any holomorphic line bundle such that $E_\pm\tensor L$ are
globally generated, let $s:=r-p+1$, and let
$\sigma_\pm:\Oo^{s}\to E_\pm\tensor L$ be $s$ general sections, with
determinantal centers
\[
D_\pm:=D_{s-1}(\sigma_\pm)=\{x:\ \rank\sigma_\pm(x)\le s-1\},
\]
of expected codimension $p=(s-(s-1))(r-(s-1))$ for general
$\sigma_\pm$ (so that Thom--Porteous applies). Then in $\CH^p(X)_{\mathbb Q}$:
\begin{equation}
\boxed{\;[D_+]-[D_-]=(-1)^{p-1}(p-1)!\,m\,\alpha\;}
\label{eq:pairlaw}
\end{equation}
--- CORRECTION-FREE: no hyperplane, binomial, or twist term survives
--- and consequently
\begin{equation}
\boxed{\;
W:=\frac{(-1)^{p-1}}{(p-1)!\,m}\bigl([D_+]-[D_-]\bigr)\in\CH^p(X)_{\mathbb Q},
\qquad \cl(W)=\alpha .}
\label{eq:output}
\end{equation}
In particular $\alpha$ is algebraic.

If moreover the pure carrier is a SINGLE exact bundle $E$
($\ch(E)=\rank+m\alpha$, hence $c_1(E)=0$) and $L=\Oo(k)$ with
$c_1(\Oo(1))^{p}=[H_1\cap\dots\cap H_p]$ a complete-intersection
class, then the single center obeys
\begin{equation}
\boxed{\;[D_E]=(-1)^{p-1}m(p-1)!\,\alpha
+\binom{r}{p}\,k^{p}\,[H_1\cap\dots\cap H_p]\;}
\label{eq:single}
\end{equation}
which at $p=2$, $m=10$ is exactly the dossier's
$-10\alpha+\binom r2k^2h^2$ bookkeeping.
\end{theorem}

\begin{proof}
By \eqref{eq:pair} and the triangularity of Newton's relations,
$c_j(E_+)=c_j(E_-)$ for $j<p$ and the virtual difference
$V:=E_+\ominus E_-$ satisfies Lemma \ref{lem:newton}, giving
$c_p(V)=(-1)^{p-1}(p-1)!\,m\,\alpha$ as a Betti class. By
Thom--Porteous for the maximal degeneracy $k=s-1$ of
$\sigma:\Oo^s\to F$ ($\rank F=r$), the determinantal center of
expected codimension $r-s+1=p$ has class
\[
[D]=c_{r-s+1}\bigl(F-\Oo^{s}\bigr)=c_p(F),
\]
since the trivial summand contributes nothing to $c(F-\Oo^s)=c(F)$.
(For $p=1$: the degeneracy divisor of $r$ sections has class
$c_1(F)$; for $\Oo\to F$ of rank $2$: the zero locus has class
$c_2(F)$; both classical.) Hence
$[D_\pm]=c_p(E_\pm\tensor L)$ in $\CH^p_{\mathbb Q}$, and by Lemma
\ref{lem:twist}
\[
[D_+]-[D_-]=c_p(E_+\tensor L)-c_p(E_-\tensor L)
=c_p(V)=(-1)^{p-1}(p-1)!\,m\,\alpha,
\]
where the last equality is the image of Lemma \ref{lem:newton} under
the canonical cycle class (algebraic Chern classes map to topological
ones). This proves \eqref{eq:pairlaw}; applying $\cl$ and scaling
gives \eqref{eq:output}, which exhibits $\alpha$ as the canonical
Betti class of the explicit rational algebraic cycle $W$.
For \eqref{eq:single}: a single pure carrier has
$\ch_j(E)=0$ for $j\notin\{0,p\}$, hence (Newton, as above)
$c_j(E)=0$ for all $j\ge1$ except $j=p$, and
$c_p(E)=(-1)^{p-1}(p-1)!\,m\alpha$. In the twist expansion only
$t=0$ and $t=p$ survive:
$c_p(E\tensor\Oo(k))=c_p(E)+\binom{r}{p}(k\,c_1(\Oo(1)))^{p}$, and
Thom--Porteous gives $[D_E]=c_p(E\tensor\Oo(k))$.
\end{proof}

\begin{corollary}[Forcing implication, typed]
\label{cor:forcing}
On arbitrary original $(X,p,\alpha)$, the single typed input of
Theorem B2(iii) --- an integrable holomorphic structure
$(\ddbar_{E_\alpha})^2=0$ on the exact carrier, equivalently a pure

thereby reduced, with zero residual ambiguity, to producing the input

\end{corollary}
